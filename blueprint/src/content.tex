\chapter{Canonical factorization in the Nevanlinna class}

\noindent\textbf{Project overview.}
This project develops a Lean 4 and mathlib formalization of Hardy spaces and
the Nevanlinna class on the unit disc. Its ultimate goal is a formal proof of
the canonical factorization theorem for the Nevanlinna class, decomposing a
nonzero function into its Blaschke, singular, and outer factors. This
blueprint records the mathematical dependency graph and the current
formalization status. Implemented proofs determine their dependencies; Garnett
supplies the proof plans for unimplemented steps. Dependencies are recorded at
roadmap-node granularity: when only part of a grouped theorem is used, the proof
identifies that part. A proof marked complete may still have open prerequisites.

\section{Chapter I: Poisson integrals and boundary values}

\begin{definition}[Poisson integral of boundary data]
  \label{def:poisson_integral}
  \lean{poissonIntegral, poissonIntegral_toSignedMeasure,
    poissonIntegral_toSignedMeasure_apply}
  \leanok
  Boundary data is a vector measure $\nu$ with values in a real normed space,
  and
  \[
    P[\nu](w)=\int P_w(z)\,d\nu(z).
  \]
  The kernel is real-valued, so it pairs with $\nu$ by scalar multiplication.
  A finite positive measure $\mu$ enters as the associated signed measure, and
  data given as a function $f$ against a reference measure $\mu$ enters as the
  vector measure $f\,d\mu$. For a finite positive measure, compatibility with
  the ordinary integral gives $P[\mu](w)=\int P_w\,d\mu$ for every $w$, without
  a support hypothesis. This follows from the integral pairing identity for
  the associated signed measure.
\end{definition}

\begin{lemma}[Basic properties of the Poisson integral]
  \label{lem:poisson_integral_linear}
  \uses{def:poisson_integral}
  \lean{poissonIntegral_neg, poissonIntegral_smul, poissonIntegral_add,
    poissonIntegral_sub, integrable_poissonKernel_toSignedMeasure,
    integrable_poissonKernel_withDensityᵥ, poissonIntegral_withDensityᵥ,
    poissonIntegral_eq_posPart_sub_negPart, integrable_poissonKernel_totalVariation,
    poissonIntegral_toSignedMeasure_sub, totalVariation_toSignedMeasure_sub_eq_zero,
    poissonIntegral_toSignedMeasure_nonneg}
  \leanok
  For vector measures, $P[-\nu]=-P[\nu]$ and $P[a\nu]=aP[\nu]$ for
  $a\in\mathbb R$. If $P_w$ is integrable against $|\nu_1|$ and $|\nu_2|$, then
  $P[\nu_1\pm\nu_2](w)=P[\nu_1](w)\pm P[\nu_2](w)$.

  Let $w\in\D(c,R)$, let $\mu$ be a finite positive measure carried by
  $\partial\D(c,R)$, and let $k\in L^1(\mu)$ be real. The kernel $P_w$ is
  integrable against $\mu$ and the variation of $k\,d\mu$, and
  \[
    P[k\,d\mu](w)=\int P_w(z)k(z)\,d\mu(z).
  \]
  For a finite signed measure $\nu$ carried by the same circle, $P_w$ is
  integrable against $|\nu|$ and its Jordan decomposition gives
  \[
    P[\nu](w)=\int P_w\,d\nu^+-\int P_w\,d\nu^-.
  \]
  Differences of finite positive measures carried by the circle are still
  carried by the circle, and their Poisson integrals subtract accordingly.
  For a positive measure, $P[\mu](w)\ge0$.
  \begin{proof}
    \proves{lem:poisson_integral_linear}
    \leanok
    Linearity follows from the vector-measure integral. At an interior point
    the kernel is bounded on the boundary circle, which gives the stated
    integrability. Split a real density into its positive and negative parts
    to obtain the density formula. Apply linearity to the Jordan decomposition
    to obtain the signed-measure formula. Positivity follows from $P_w\ge0$
    on the boundary circle.
  \end{proof}
\end{lemma}

\begin{theorem}[Poisson integrals of boundary data are harmonic]
  \label{thm:harmonic_poisson_integral}
  \uses{def:poisson_integral}
  \lean{harmonicOnNhd_circleAverage_poissonKernel_smul,
    harmonicOnNhd_poissonIntegral_toSignedMeasure,
    harmonicOnNhd_poissonIntegral_withDensityᵥ}
  \leanok
  For a finite positive measure $\mu$ carried by the circle and real
  $k\in L^1(\mu)$, both $P[\mu]$ and $P[k\,d\mu]$ are harmonic inside.
  If $f$ is continuous on $\partial\D(c,R)$ then $P[f\,d\sigma]$ is harmonic on
  the open disc, where $\sigma$ is normalized arclength.
  \begin{proof}
    \proves{thm:harmonic_poisson_integral}
    \uses{lem:poisson_integral_linear}
    \leanok
    For a positive measure, take the real part of its analytic
    Herglotz--Riesz transform. For a real density, split it into its positive
    and negative parts and use linearity.
    Harmonicity of the circle average follows from harmonicity of the kernel in
    the interior variable, by differentiating under the integral sign.
  \end{proof}
\end{theorem}

\begin{theorem}[Poisson integral formula]
  \label{thm:poisson_reproducing}
  \uses{def:poisson_integral}
  \lean{poissonIntegral_circleMeasure_eq_self, poissonIntegral_circleMeasure_eq_one,
    integral_poissonKernel_smul_circleMeasure_eq_circleAverage,
    poissonIntegral_circleMeasure_withDensityᵥ_eq_circleAverage}
  \leanok
  If $u$ is harmonic on a neighbourhood of $\overline{\D}(c,R)$ then
  $u(w)=P[u\,d\sigma_{c,R}](w)$ for every interior $w$. Taking $u\equiv 1$ gives
  $P[\sigma_{c,R}](w)=1$: the kernel has total mass one at every interior point.
  \begin{proof}
    \proves{thm:poisson_reproducing}
    \uses{lem:poisson_integral_linear}
    \leanok
    Transport to the circle average and apply the mathlib form of the Poisson
    formula. Total mass one is the case $u\equiv 1$, read through the density
    lemma.
  \end{proof}
\end{theorem}

\begin{theorem}[Dirichlet problem on a disc]
  \label{thm:poisson_extension}
  \uses{def:poisson_integral}
  \lean{poissonIntegral_continuousOn_closedBall_eq_boundary}
  \leanok
  For continuous real boundary data $u$ on $\partial\D(c,R)$ there is a function
  continuous on $\overline{\D}(c,R)$, harmonic inside, equal to
  $P[u\,d\sigma_{c,R}]$ on the open disc and to $u$ on the circle.

  \begin{proof}
    \proves{thm:poisson_extension}
    \uses{thm:harmonic_poisson_integral, thm:poisson_reproducing}
    \notready
    The Poisson kernel has mass one and its mass away from a boundary
    point tends uniformly to zero. Uniform continuity of the boundary data
    gives continuity of the extension; interior harmonicity is already
    formalized. The continuous-extension declaration still has an open proof.
  \end{proof}
\end{theorem}

\begin{theorem}[A boundary measure is determined by its Poisson integral]
  \label{thm:poisson_measure_unique}
  \uses{def:poisson_integral}
  \lean{eq_of_eqOn_poissonIntegral}
  \leanok
  \mainline
  \bookref{Garnett}{I.3, after Theorem 3.1}{Printed page 15, PDF page 27}
  If $\mu$ and $\nu$ are finite signed measures carried by $\partial\D(c,R)$,
  $R>0$, with $P[\mu]=P[\nu]$ on $\D(c,R)$, then $\mu=\nu$.
  \begin{proof}
    \proves{thm:poisson_measure_unique}
    \uses{thm:poisson_extension}
    \notready
    The standard weak-star argument: $P[\mu](r\,\cdot)\,d\sigma\to\mu$ when
    tested against continuous functions on the circle, because the Poisson
    integral of continuous data converges uniformly to it. This is what
    licenses speaking of \emph{the} representing measure in
    Theorem~\ref{thm:II_5_3}, and it is used again in the uniqueness part of
    the canonical factorization
    (Theorem~\ref{thm:nevanlinna_factorization_uniqueness}) and in Garnett's
    Theorem II.5.4.
  \end{proof}
\end{theorem}

\begin{definition}[Nontangential approach region]
  \label{def:nt_region}
  \lean{Complex.stolzSetAt}
  \leanok
  \mainline
  \bookref{Garnett}{I.4}{Chapter I, Section 4, printed page 21, PDF page 33, disc form}
  For $\zeta\in\partial\D$ and $\alpha>1$ put
  \[
    \Gamma_\alpha(\zeta)
      =\Bigl\{z\in\D:\ |\zeta-z|<\alpha\,(1-|z|)\Bigr\},
  \]
  a teardrop-shaped region asymptotic at $\zeta$ to an angle with vertex
  $\zeta$. Facts to record: $\Gamma_\alpha(\zeta)$ is open and contained in
  $\D$; it increases with $\alpha$; it contains the radial segment
  $[0,1)\zeta$; it is empty when $\alpha\le 1$; and
  $\Gamma_\alpha(1)$ is mathlib's \texttt{Complex.stolzSet}~$\alpha$, with
  $\Gamma_\alpha(\zeta)=\zeta\cdot\Gamma_\alpha(1)$. Mathlib only has the
  region at $\zeta=1$, so the rotated family has to be added.
\end{definition}

\begin{definition}[Nontangential limit]
  \label{def:nt_limit}
  \lean{HasNontangentialLimitWithin, nontangentially, HasNontangentialLimit,
    NontangentiallyConvergentAt, hasNontangentialLimit_iff_forall}
  \leanok
  \uses{def:nt_region}
  \mainline
  \bookref{Garnett}{I.5}{Chapter I, Section 5, printed page 27, PDF page 39}
  Write $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$ for the neighbourhood filter
  of $\zeta$ within $\Gamma_\alpha(\zeta)$. The \emph{nontangential filter} at
  $\zeta$ is their supremum in the lattice of filters,
  \[
    \mathcal{N}_\zeta=\sup_{\alpha>1}\ \mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta).
  \]
  For $\alpha>1$, $f$ has limit $L$ at $\zeta$ \emph{within} $\Gamma_\alpha(\zeta)$
  if $f\to L$ along $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$, and $f$ has
  \emph{nontangential limit} $L$ at $\zeta$ if $f\to L$ along
  $\mathcal{N}_\zeta$. The two are related by
  \[
    f\to L \ \text{along}\ \mathcal{N}_\zeta
    \iff
    \forall\,\alpha>1,\ f\to L \ \text{along}\ \mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta).
  \]
\end{definition}

\begin{definition}[Boundary value]
  \label{def:boundary_function}
  \lean{boundaryValue, IsAEBoundaryValue}
  \leanok
  \uses{def:nt_limit}
  The boundary value of $f$ at $\zeta$ is
  \[
    f^*(\zeta)=\lim_{\mathcal{N}_\zeta}f,
  \]
  the limit along the nontangential filter $\mathcal{N}_\zeta$, read through the
  choice operator so that it is a total function. Where the nontangential limit
  at $\zeta$ exists, it equals $f^*(\zeta)$.
\end{definition}

\begin{lemma}[The nontangential filter is nontrivial]
  \label{lem:nt_filter_neBot}
  \uses{def:nt_region, def:nt_limit}
  \lean{radial_mem_stolzSetAt, nontangentially_neBot_of_norm_eq_one,
    ae_nontangentially_neBot_circleMeasure}
  \leanok
  For $\zeta\in\partial\D$ and $\alpha>1$ the radial segment $[0,1)\zeta$ lies in
  $\Gamma_\alpha(\zeta)$; hence $\mathcal{N}_\zeta\neq\bot$.
  \begin{proof}
    \proves{lem:nt_filter_neBot}
    \leanok
    For $0\le r<1$ one has $|r\zeta|=r<1$ and
    $|\zeta-r\zeta|=1-r<\alpha(1-r)$. So $\zeta$ lies in the closure of
    $\Gamma_\alpha(\zeta)$, making $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$
    nontrivial; a supremum of filters is nontrivial as soon as one of them is,
    so a single $\alpha$ suffices.
  \end{proof}
\end{lemma}

\begin{lemma}[Boundary values: pointwise, relational and canonical]
  \label{lem:boundary_value_bridge}
  \uses{def:boundary_function}
  \lean{NontangentiallyConvergentAt.hasNontangentialLimit,
    HasNontangentialLimit.boundaryValue_eq,
    IsAEBoundaryValue.ae_eq_boundaryValue,
    IsAEBoundaryValue.aestronglyMeasurable_boundaryValue}
  \leanok
  Let $\mu$ be a measure on $\partial\D$ carried by the circle.
  \begin{enumerate}
    \item If $f$ converges nontangentially at $\zeta$, then $f\to f^*(\zeta)$
      along $\mathcal{N}_\zeta$.
    \item If $f$ has nontangential limit $L$ at $\zeta$, then $f^*(\zeta)=L$.
    \item If $g$ is an almost everywhere nontangential boundary value of $f$,
      then $g=f^*$ almost everywhere.
    \item Consequently $f^*$ is almost everywhere strongly measurable as soon as
      one such $g$ is.
  \end{enumerate}
  Part (3) is why measurability of a boundary function never needs a separate
  argument: it is inherited from whichever explicit representative the relevant
  existence theorem produces.
  \begin{proof}
    \proves{lem:boundary_value_bridge}
    \uses{lem:nt_filter_neBot}
    \leanok
    (1) is the choice of the canonical limit. For (2), uniqueness of limits
    uses the nontrivial filter $\mathcal{N}_\zeta$. Apply (2) almost
    everywhere for (3), and transfer measurability along that equality for (4).
  \end{proof}
\end{lemma}

\begin{lemma}[Poisson kernel over a Stolz region]
  \label{lem:poisson_stolz_bound}
  \uses{def:nt_region}
  \lean{poissonKernel_le_of_mem_stolzSetAt}
  \leanok
  Let $\alpha>1$. There is a constant $C_\alpha$ such that for every
  $\zeta\in\partial\D$, every $z\in\Gamma_\alpha(\zeta)$ and every
  $\theta$,
  \[
    P_z(\theta)\ \le\
      \frac{C_\alpha\,(1-|z|)}{\bigl((1-|z|)+|e^{i\theta}-\zeta|\bigr)^{2}} .
  \]
  In particular $P_z(\theta)\le C_\alpha/(1-|z|)$ always, and
  $P_z(\theta)\le C_\alpha (1-|z|)/|e^{i\theta}-\zeta|^{2}$ once
  $|e^{i\theta}-\zeta|\ge 2\alpha(1-|z|)$. Together with
  $\int P_z\,d\theta/2\pi=1$ this exhibits the Poisson kernels, restricted to a
  Stolz region, as an approximate identity with a uniformly integrable
  decreasing majorant. This single estimate replaces Garnett's Theorem I.4.2
  and is the engine of both Lemma~\ref{lem:I_5_4} and
  Theorem~\ref{thm:I_5_3_ac}.
  \begin{proof}
    \proves{lem:poisson_stolz_bound}
    \uses{thm:poisson_reproducing}
    \leanok
    From $|e^{i\theta}-z|\ge 1-|z|$ and, by the triangle inequality and the
    definition of $\Gamma_\alpha(\zeta)$,
    $|e^{i\theta}-z|\ge|e^{i\theta}-\zeta|-\alpha(1-|z|)$, one gets
    $|e^{i\theta}-z|\ge c_\alpha\bigl((1-|z|)+|e^{i\theta}-\zeta|\bigr)$;
    insert this into $P_z(\theta)=(1-|z|^2)/|e^{i\theta}-z|^{2}$ and use
    $1-|z|^2\le 2(1-|z|)$.
  \end{proof}
\end{lemma}

\begin{theorem}[Differentiation of measures on the circle]
  \label{thm:circle_differentiation}
  \lean{Besicovitch.ae_tendsto_rnDeriv,
    VitaliFamily.ae_eventually_measure_zero_of_singular,
    Besicovitch.tendsto_filterAt,
    ae_tendsto_average_abs_sub_circleMeasure,
    norm_circleMap_sub_circleMap, abs_sub_le_norm_circleMap_sub,
    circleMeasure_closedBall_le_of_norm_eq_one}
  \leanok
  Put $\sigma=\texttt{circleMeasure }0\ 1$. If $\nu$ is a finite positive
  Borel measure on $\C$, then for $\sigma$-almost every $\zeta$,
  \[
    \lim_{h\downarrow0}
      \frac{\nu\bigl(\overline B(\zeta,h)\bigr)}
           {\sigma\bigl(\overline B(\zeta,h)\bigr)}
      =\frac{d\nu}{d\sigma}(\zeta),
  \]
  and in particular the quotient tends to zero almost everywhere when
  $\nu\perp\sigma$. Moreover, almost every $\zeta$ is a Lebesgue point of a
  given $k\in L^1(\sigma)$:
  \[
    \lim_{h\downarrow0}\ \frac{1}{\sigma(\overline B(\zeta,h))}
      \int_{\overline B(\zeta,h)}|k-k(\zeta)|\,d\sigma\ =\ 0.
  \]
  For $\|\zeta\|=1$ the supporting chord--arc estimate also gives
  $\sigma(\overline B(\zeta,h))\le \operatorname{ofReal}(h/2)$.
  \begin{proof}
    \proves{thm:circle_differentiation}
    \leanok
    Apply mathlib's Besicovitch Vitali family directly on $\C$ and transport
    its filter to shrinking closed balls. Its singular-measure theorem gives
    the zero-density assertion, and its differentiation theorem gives the
    Radon--Nikodym derivative and Lebesgue points. The chord--arc estimate is
    Jordan's sine inequality applied to the circle parametrization.
  \end{proof}
\end{theorem}

\begin{lemma}[Uniform ball-density bound for positive Poisson integrals]
  \label{lem:poisson_ball_density_bound}
  \uses{def:poisson_integral}
  \lean{integral_poissonKernel_le_of_measure_closedBall_le}
  \leanok
  Let $\nu$ be a finite positive measure carried by $\partial\D$, let
  $\|\zeta\|=1$, and suppose that for every $r>0$,
  \[
    \nu(\overline B(\zeta,r))
      \le c\,\sigma(\overline B(\zeta,r)).
  \]
  Then for $z\in\Gamma_\alpha(\zeta)$,
  \[
    \int_{\partial\D}P_z\,d\nu\le 8(\alpha+2)^2c.
  \]
  \begin{proof}
    \proves{lem:poisson_ball_density_bound}
    \uses{lem:poisson_stolz_bound, thm:circle_differentiation}
    \leanok
    Decompose the circle into dyadic chordal annuli centred at $\zeta$: with
    $\delta=1-|z|$, put $S_0=\overline B(\zeta,\delta)$ and
    $S_j=\overline B(\zeta,2^j\delta)\setminus\overline B(\zeta,2^{j-1}\delta)$.
    On $S_j$ the Stolz-region kernel bound gives
    $P_z\le 4A/(4^j\delta)$ with $A=2(\alpha+2)^2$, while the density
    hypothesis and the chord--arc estimate give
    $\nu(S_j)\le c\,2^j\delta/2$; the product is $2Ac\,2^{-j}$ and
    $\sum_j 2^{-j}=2$, so the total is $4Ac=8(\alpha+2)^2c$, exactly the
    stated constant.

    In Lean the estimate is carried out for the lower integral, so that no
    piecewise integrability is needed, and transferred at the end by
    \texttt{integral\_eq\_lintegral\_of\_nonneg\_ae}. Using
    \texttt{lintegral\_iUnion\_le} rather than \texttt{lintegral\_iUnion}
    means only that the annuli cover the plane has to be proved, not that they
    are pairwise disjoint.
  \end{proof}
\end{lemma}

\begin{lemma}[Density-zero Poisson integrals]
  \label{lem:poisson_density_zero}
  \uses{def:nt_limit, def:poisson_integral}
  \lean{hasNontangentialLimit_integral_poissonKernel_of_density_zero}
  \leanok
  Let $\nu$ be a finite positive measure carried by $\partial\D$. If
  \[
    \frac{\nu(\overline B(\zeta,r))}
         {\sigma(\overline B(\zeta,r))}\longrightarrow0
    \qquad(r\downarrow0),
  \]
  then $P[\nu](z)\to0$ nontangentially as $z\to\zeta$.
  \begin{proof}
    \proves{lem:poisson_density_zero}
    \uses{lem:poisson_ball_density_bound, lem:poisson_stolz_bound}
    \leanok
    Restrict $\nu$ to a small fixed ball about $\zeta$ and its complement.
    Density zero supplies a uniform ball-density bound for the near part, so
    Lemma~\ref{lem:poisson_ball_density_bound} makes its Poisson integral
    arbitrarily small. The far part stays a fixed positive distance from
    $\zeta$, and its Poisson integral tends uniformly to zero in every Stolz
    region.
  \end{proof}
\end{lemma}

\begin{lemma}[Garnett I.5.4: singular Poisson integrals vanish at the boundary]
  \label{lem:I_5_4}
  \uses{def:nt_limit, def:poisson_integral}
  \lean{isAEBoundaryValue_poissonIntegral_zero_of_mutuallySingular}
  \leanok
  \mainline
  \bookref{Garnett}{I.5.4}{Chapter I, Section 5, PDF pages 41--42}
  If $\nu$ is a finite signed measure on $\partial\D$ that is singular with
  respect to $d\theta$, then
  \[
    \operatorname{nt}\lim_{z\to\zeta}P[\nu](z)=0
  \]
  for almost every $\zeta\in\partial\D$. This is the step that removes the
  singular part in Theorem~\ref{thm:I_5_3}, and it is also what gives the
  singular functions of Definition~\ref{def:singular_function_formula} boundary
  modulus one almost everywhere.
  \begin{proof}
    \proves{lem:I_5_4}
    \uses{lem:poisson_density_zero, thm:circle_differentiation, lem:poisson_integral_linear}
    \leanok
    Mutual singularity makes the total variation of $\nu$ singular with
    respect to $\sigma$. Expanding the Jordan decomposition and using
    subadditivity of vector-measure variation gives
    $\operatorname{var}(\nu)\le |\nu|$, so the variation measure is also
    carried by the circle and singular to $\sigma$. Theorem~\ref{thm:circle_differentiation}
    gives density zero almost everywhere, and Lemma~\ref{lem:poisson_density_zero}
    gives $P[\operatorname{var}(\nu)]\to0$. Finally
    $|P[\nu]|\le P[\operatorname{var}(\nu)]$ follows from the vector-integral
    norm inequality, and squeezing proves the claim.
  \end{proof}
\end{lemma}

\begin{theorem}[Fatou at Lebesgue points]
  \label{thm:I_5_3_ac}
  \uses{def:nt_limit, def:poisson_integral}
  \lean{hasNontangentialLimit_poissonIntegral_of_lebesguePoint,
    isAEBoundaryValue_poissonIntegral}
  \leanok
  Let $k\in L^1(d\theta/2\pi)$. Then at every Lebesgue point $\zeta$ of $k$,
  and hence almost everywhere,
  \[
    \operatorname{nt}\lim_{z\to\zeta}P\bigl[k\,d\theta/2\pi\bigr](z)=k(\zeta).
  \]
  \begin{proof}
    \proves{thm:I_5_3_ac}
    \uses{lem:poisson_density_zero, thm:circle_differentiation, thm:poisson_reproducing,
      lem:poisson_integral_linear}
    \leanok
    At a Lebesgue point put
    $d\rho_\zeta=|k-k(\zeta)|\,d\sigma$. Its ball-density quotient is exactly
    the mean oscillation of $k$, hence tends to zero. Apply
    Lemma~\ref{lem:poisson_density_zero} to $\rho_\zeta$, use
    $\int P_z\,d\sigma=1$, and bound
    $|P[k](z)-k(\zeta)|$ by $P[\rho_\zeta](z)$. The almost-everywhere statement
    follows from Theorem~\ref{thm:circle_differentiation}.
  \end{proof}
\end{theorem}

\begin{theorem}[Garnett I.5.3: Fatou theorem, $p=1$ disc form]
  \label{thm:I_5_3}
  \uses{def:nt_limit, def:boundary_function, def:poisson_integral}
  \lean{isAEBoundaryValue_poissonIntegral_rnDeriv,
    boundaryValue_poissonIntegral, integrable_boundaryValue_poissonIntegral}
  \leanok
  \mainline
  \bookref{Garnett}{I.5.3}{Chapter I, Section 5, PDF pages 40--41}
  Let $\mu$ be a finite signed measure on $\partial\D$, with Lebesgue
  decomposition $d\mu=k\,d\theta/2\pi+d\mu_s$, $\mu_s\perp d\theta$. Then the
  Poisson integral of $\mu$ has a nontangential limit at almost every boundary
  point, and
  \[
    \operatorname{nt}\lim_{z\to\zeta}P[\mu](z)=k(\zeta)
    \qquad\text{for almost every }\zeta\in\partial\D;
  \]
  equivalently $\bigl(P[\mu]\bigr)^*=k$ almost everywhere, so the boundary
  function is in $L^1$. Garnett states the stronger half-plane theorem for all
  $1\le p\le\infty$; only this $p=1$ disc form is used in this project.
  \begin{proof}
    \proves{thm:I_5_3}
    \uses{thm:I_5_3_ac, lem:I_5_4, lem:poisson_integral_linear, lem:boundary_value_bridge}
    \leanok
    Split $\mu$ by the Lebesgue decomposition, which mathlib provides as
    \texttt{MeasureTheory.SignedMeasure.singularPart\_add\_withDensity\_rnDeriv\_eq}, and add
    Theorem~\ref{thm:I_5_3_ac} to Lemma~\ref{lem:I_5_4}, using that
    nontangential limits are limits along a filter and therefore additive.
    The density part is identified with its signed Poisson integral by
    Lemma~\ref{lem:poisson_integral_linear}; integrability of the kernel
    against both summands then identifies their sum with $P[\mu]$. Finally use
    Lemma~\ref{lem:boundary_value_bridge} for the canonical boundary function.
  \end{proof}
\end{theorem}

\section{Chapter I: harmonic representation}

\begin{lemma}[Garnett I.(3.5): Poisson integrals contract $L^p$, disc form]
  \label{eq:I_3_5}
  \uses{def:poisson_integral}
  \lean{eLpNorm_poissonIntegral_radial_le, hardyNorm_poissonIntegral_le}
  \leanok
  \bookref{Garnett}{I.(3.5)}{Chapter I, Section 3, printed pages 14--15,
    PDF pages 26--27}
  Let $1\le p\le\infty$, let $\sigma$ be normalized arclength on $\partial\D$,
  and let $U\in L^p(\sigma)$. Then for $0<r<1$,
  \[
    \|P[U\,d\sigma](r\,\cdot)\|_{L^p(\sigma)}\le\|U\|_{L^p(\sigma)},
  \]
  hence $\sup_{0<r<1}\|P[U\,d\sigma](r\,\cdot)\|_{L^p(\sigma)}
  \le\|U\|_{L^p(\sigma)}$.
  \begin{proof}
    \proves{eq:I_3_5}
    \uses{lem:poisson_integral_linear, thm:poisson_reproducing}
    \leanok
    For $p=\infty$ use $P_z\ge0$ and $\int P_z\,d\sigma=1$. For
    $1\le p<\infty$, Jensen's inequality for the probability measure
    $P_z\,d\sigma$ gives $|P[U\,d\sigma](z)|^p\le\int P_z|U|^p\,d\sigma$.
    Integrate over $z=r\eta$, exchange the integrals, and use
    $P_{r\eta}(\zeta)=P_{r\zeta}(\eta)$ for $|\eta|=|\zeta|=1$ together with
    total mass one. Garnett uses Minkowski's integral inequality instead,
    which mathlib lacks.
  \end{proof}
\end{lemma}

\begin{theorem}[Garnett I.3.5(a): bounded radial $L^p$ norms, disc form]
  \label{thm:I_3_5(a)}

  \lean{harmonicOnNhd_hardyNorm_lt_top_iff_exists_memLp_eq_poissonIntegral,

  \leanok

  A function $u:\mathbb C\to\mathbb R$ is harmonic on $\D$ and satisfies
  \[
    \sup_{0<r<1}\|u(r\,\cdot)\|_{L^p(\sigma)}<\infty
  \]

  $u=P[U\,d\sigma]$ on $\D$.

    \proves{thm:I_3_5(a)}
    \uses{lem:poisson_integral_linear, thm:harmonic_poisson_integral, thm:poisson_reproducing,

    point $U$ as $t\to1$. Testing against the fixed kernel $P_z\in L^q$
    gives $u=P[U\,d\sigma]$. Conversely, the Poisson operators are
    contractions on $L^p$ (Lemma~\ref{eq:I_3_5}).
    The Lean proof is complete relative to the finite-measure dual representation
    theorem in Theorem~\ref{thm:lp_duality}, whose existence proof remains open.
  \end{proof}

\begin{theorem}[Garnett I.6.3: harmonic comparison characterization]
  \label{thm:I_6_3}
  \notready
  \bookref{Garnett}{I.6.3}{Printed pages 34--35, PDF pages 46--47}
  An upper semicontinuous $v:\Omega\to[-\infty,\infty)$ is subharmonic if
  and only if, for every bounded open $W$ compactly contained in $\Omega$
  and every harmonic $u$ on $W$, the boundary condition
  $\limsup_{W\ni z\to\zeta}(v(z)-u(z))\le0$ implies $v\le u$ on $W$.

    \proves{thm:I_6_3}
    \uses{thm:harmonic_mean_value, thm:poisson_extension}
    \notready

    upper semicontinuous boundary data from above by continuous functions,
    solve the Dirichlet problems, and pass to the circular mean inequality;
    integrating in the radius gives the ball mean inequality.
    The active Lean definition uses comparison on balls; the equivalence
    with the book's mean-value definition remains to be formalized.
  \end{proof}
\end{theorem}

\begin{corollary}[Garnett I.6.4: integrability on interior circles]
  \label{cor:I_6_4}
  \notready
  \bookref{Garnett}{I.6.4}{Printed page 35, PDF page 47}
  If $v$ is subharmonic on a connected open set $\Omega$ and
  $v\not\equiv-\infty$, then its restriction to every circle bounding a

\begin{theorem}[Garnett I.6.5: Poisson modification]

\begin{lemma}[Harnack's inequality on a disc]
  \label{lem:harnack_inequality}
  \lean{InnerProductSpace.harnack_inequality, InnerProductSpace.harnack_le,

  on $\partial\D(c,R)$. Then for $w$ with $\rho=|w-c|<R$,

  $\frac{R-\rho}{R+\rho}\le P_w(\zeta)\le\frac{R+\rho}{R-\rho}$ underlying it is
  recorded separately, since it is used on its own.

    \uses{thm:poisson_reproducing}
    \leanok

    Herglotz--Riesz kernel. Insert them into the Poisson integral formula
    $u(w)=\int P_w u\,d\sigma$, using $u\ge0$ on the circle to preserve the
    inequalities, and identify $\int u\,d\sigma$ with $u(c)$ by the mean value
    property.
  \end{proof}
\end{lemma}

\begin{theorem}[Harnack's convergence theorem]
  \label{thm:harnack_convergence}
  \notready
  \bookref{Garnett}{I.6.7, proof}{Invoked on printed page 37, PDF page 49}
  Let $u_n$ be an increasing sequence of real harmonic functions on a
  connected domain. If $\sup_n u_n(z_0)<\infty$ at one interior point,
  then $u_n$ converges locally uniformly to a finite harmonic function.
  \begin{proof}
    \proves{thm:harnack_convergence}
    \uses{lem:harnack_inequality, thm:harmonic_mean_value}
    \notready
    Apply Harnack's inequality to the nonnegative differences $u_m-u_n$;
    convergence at one point gives uniform Cauchy estimates on compact
    subsets by a chain of discs. Pass the mean-value identity to the local
    uniform limit. For Poisson modifications, apply this on each smaller
    disc to a cofinal sequence of radii.
  \end{proof}
\end{theorem}

\begin{theorem}[Harmonic majorants of subharmonic functions, Garnett I.6.7]
  \label{thm:I_6_7}
  \lean{Subharmonic.hasHarmonicMajorant_iff_radialMean_bddAbove,
    Subharmonic.exists_isLeastHarmonicMajorant_tendsto_poissonModification}
  \leanok
  \mainline
  \bookref{Garnett}{I.6.7}{Chapter I, Section 6, printed pages 36--37, PDF pages 48--49}
  Let $v$ be subharmonic on $\mathbb{D}$. Then $v$ has a harmonic majorant on
  $\mathbb{D}$ if and only if
  \[
    \sup_{0<r<1}\frac{1}{2\pi}
      \int_0^{2\pi}v(r e^{i\theta})\,d\theta
    = \sup_{0<r<1} v_r(0) < \infty,
  \]
  where $v_r$ is the Poisson extension to $|z|<r$ of the boundary values of
  $v$ on $|z|=r$. When this condition holds and $v\not\equiv-\infty$, the
  least harmonic majorant $u$ of $v$ is given by
  \[
    u(z)=\lim_{r\to 1}
      \frac{1}{2\pi}\int_0^{2\pi}
        P_{z/r}(\theta)v(r e^{i\theta})\,d\theta
      =\lim_{r\to 1}v_r(z).
  \]
  \begin{proof}
    \proves{thm:I_6_7}
    \uses{thm:I_6_5, thm:harnack_convergence, thm:I_6_3}
    \notready
    The modifications $v_r$ increase and are harmonic on expanding discs.
    If their values at zero are bounded, Harnack's convergence theorem
    gives a finite harmonic limit $u\ge v$. Every harmonic majorant $U$
    dominates every $v_r$ by comparison, hence dominates $u$; this proves
    leastness. Conversely $v_r\le U$ bounds their values at zero.
    The case $v\equiv-\infty$ has harmonic majorants but no least one and
    is excluded from the second assertion.
  \end{proof}
\end{theorem}

\begin{definition}[Multiplicity and the Blaschke sum]
  \label{def:blaschke_sum}
  \lean{blaschkeSum}
  \leanok
  For $m:\mathbb C\to\mathbb N\cup\{\infty\}$, define
  \[
    \mathcal B(m)=\sum_{a\in\mathbb D}m(a)(1-|a|)\in[0,+\infty].
  \]
  An analytic function uses $m_f(a)=\operatorname{ord}_a(f)$; a family
  $(a_i)_{i\in I}$ uses $m(a)=\#\{i\in I:a_i=a\}$, with infinite cardinalities
  recorded as $\infty$.
\end{definition}

\begin{definition}[Blaschke condition for a multiplicity function]
  \label{def:blaschke_condition}
  \lean{BlaschkeCondition, BlaschkeCondition.ne_top}
  \leanok
  \uses{def:blaschke_sum}
  A multiplicity function $m$ satisfies Blaschke's condition on $\mathbb D$
  if $\mathcal B(m)<\infty$. In that case $m(a)<\infty$ for every $a\in\mathbb D$.
\end{definition}

\begin{definition}[Lean Blaschke factors]
  \label{def:blaschke_factor}
  \lean{BlaschkeFactor, normedBlaschkeFactor}
  \leanok
  For $R>0$ and $|a|<R$, the elementary Blaschke factor is
  \[
    b_{R,a}(z)=\frac{R(z-a)}{R^2-\overline a z}.
  \]
  Its normalized form is $b_{R,0}$ when $a=0$, and
  $-\overline a\,b_{R,a}/|a|$ otherwise.
\end{definition}

\begin{definition}[Blaschke product of a multiplicity function]
  \label{def:blaschke_product}
  \lean{blaschkeProductFactor, BlaschkeProduct}
  \leanok
  \uses{def:blaschke_factor, def:blaschke_sum}
  For $m:\mathbb C\to\mathbb N\cup\{\infty\}$, with $b_a$ the normalized factor
  $b_{1,a}$,
  \[
    B_m(z)=z^{m(0)}\prod_{a\in\D\setminus\{0\}}b_a(z)^{m(a)},
  \]
  the product taken unconditionally, i.e.\ as the limit along finite subsets of $\D$.
  An analytic function uses $m=\operatorname{ord}(f)$ and a finite or infinite
  family $(a_i)$ uses $m(a)=\#\{i:a_i=a\}$. Points outside $\D$ are ignored.
\end{definition}

\begin{theorem}[Garnett II.2.1: the Blaschke condition]
  \label{thm:II_2_1}
  \uses{def:blaschke_condition}
  \lean{blaschkeCondition_of_hasHarmonicMajorant,

    \mathcal B(m_f)\le u(0)-\log|f(0)|.

    \uses{thm:I_6_7, thm:log_abs_subharmonic}

  \end{proof}

\begin{theorem}[Garnett II.2.2: convergence and boundary size of Blaschke products]
  \label{thm:II_2_2}
  \uses{def:blaschke_product, def:blaschke_condition, def:boundary_function}
  \lean{tendstoLocallyUniformlyOn_blaschkeProductFactor,
    tendstoLocallyUniformlyOn_blaschkeProduct,
    analyticOnNhd_blaschkeProduct, norm_blaschkeProduct_le_one,
    norm_BlaschkeFactor_one_le_one,
    analyticOrderAt_blaschkeProduct, blaschkeProduct_eq_zero_iff,
    memHpDisc_blaschkeProduct, ae_hasNontangentialLimit_blaschkeProduct}
  \leanok
  \mainline
  \bookref{Garnett}{II.2.2}{Chapter II, Section 2, PDF page 64}
  If $m$ satisfies the Blaschke condition, then the finite partial products
  of $B_m$ converge uniformly on compact subsets of $\mathbb{D}$ to a bounded
  analytic function whose zeros are exactly the points with $m(w)>0$, with
  the prescribed multiplicities, and whose boundary modulus is one almost
  everywhere. More precisely, $|B_m|\le1$ on $\D$ and
  $\operatorname{ord}_w B_m=m(w)$ for every $w\in\D$.
  Its extension by zero outside $\D$ belongs to $H^\infty(\D)$.
  Almost everywhere on the circle, its canonical boundary value is its
  nontangential limit and has modulus one.
  \begin{proof}
    \proves{thm:II_2_2}
    \uses{cor:II_3_2(p=∞)}
    \notready
    The passage from the product omitting origin zeros to the full product
    $B(z)=z^mP(z)$ is formalized: multiplication by $z^m$, whose modulus
    is at most one on $\D$, preserves locally uniform convergence.
    Analyticity of $B$ is formalized from this convergence: the finite
    products are holomorphic since $|\overline{a}z|<1$ for $a,z\in\D$,
    and a locally uniform limit of holomorphic functions is holomorphic.
    The bound $|B|\le1$ is formalized by bounding each finite product
    and passing to the pointwise limit. The factor bound follows from
    $|1-\overline a z|^2-|z-a|^2=(1-|a|^2)(1-|z|^2)\ge0$.
    Membership in $H^\infty$ follows from analyticity and $|B|\le1$, and
    the zero set is read off from the multiplicity formula, since an analytic
    function vanishes exactly where its order is nonzero.
    Convergence of the product omitting origin zeros, the multiplicity
    formula, and the boundary modulus remain to be proved.
    For the boundary modulus, first remove the origin zeros and let $B_n$
    be the finite partial products. The tail product $Q_n=B/B_n$ is bounded
    analytic. Corollary~\ref{cor:II_3_2(p=∞)} gives
    \[
      |Q_n(0)|\le\int_{\partial\D}|Q_n^*|\,d\sigma
      =\int_{\partial\D}|B^*|\,d\sigma\le1.
    \]
    Here $|B_n|=1$ on the circle and $|Q_n(0)|\to1$, so $|B^*|=1$
    almost everywhere. Restoring the origin factor preserves the boundary modulus.
  \end{proof}
\end{theorem}

\begin{corollary}[II.3.1, $p=\infty$: boundary values on the disc]

  \begin{proof}
    \proves{cor:II_3_1(p=∞)}
    \uses{thm:I_3_5(a), thm:I_5_3_ac}
    \leanok
    The real and imaginary parts of $f$ are bounded harmonic, hence Poisson
    integrals of $L^\infty$ boundary data by Theorem~\ref{thm:I_3_5(a)} with
    $p=\infty$. These densities lie in $L^1$, so Theorem~\ref{thm:I_5_3_ac}
    gives their nontangential limits almost everywhere. Intersect the two
    full-measure sets and combine the real and imaginary limits.
    The $H^\infty$ case uses the equality of the Hardy norm with the pointwise supremum.
    Both Lean proofs are complete; I.3.5(a) remains an unproved prerequisite.
  \end{proof}
\end{corollary}

\begin{corollary}[Garnett II.3.2, $p=\infty$: Poisson representation on the disc]
  \label{cor:II_3_2(p=∞)}
  \uses{def:poisson_integral, def:boundary_function}
  \notready
  \bookref{Garnett}{II.3.2 ($p=\infty$)}{Chapter II, Section 3,
    printed page 56, PDF page 68; disc form}
  Let $\sigma$ be normalized arclength on $\partial\D$ and let
  $U\in L^\infty(\sigma;\mathbb C)$. Then $U$ is the a.e. nontangential
  boundary value of an $H^\infty(\D)$ function if and only if
  $P[U\,d\sigma]$ is analytic on $\D$. Every such function equals
  $P[U\,d\sigma]$ on $\D$. In particular, for $f\in H^\infty(\D)$,
  \[
    f(z)=P[f^*\,d\sigma](z),\qquad
    |f(0)|\le\int_{\partial\D}|f^*|\,d\sigma.
  \]
  \begin{proof}
    \proves{cor:II_3_2(p=∞)}
    \uses{thm:I_3_5(a), thm:I_5_3_ac, eq:I_3_5}
    \notready
    Apply Theorem~\ref{thm:I_3_5(a)} to the real and imaginary parts of
    a bounded analytic function. Theorem~\ref{thm:I_5_3_ac} identifies the
    resulting densities with its boundary values. Conversely, positivity
    and unit mass of the Poisson kernel give the $L^\infty$ bound, and
    Theorem~\ref{thm:I_5_3_ac} gives the prescribed boundary values.
    At the origin the kernel is one, giving the displayed inequality.
  \end{proof}
\end{corollary}

\begin{theorem}[Garnett II.4.1: logarithmic inequality, $H^\infty$ disc case]
  \label{thm:II_4_1}
  \uses{def:boundary_function, def:poisson_integral}
  \notready
  \mainline
  \bookref{Garnett}{II.4.1 ($p=\infty$)}{Printed page 62, PDF page 74}

  $\log|f^*|\in L^1(\sigma)$, with $\log0=-\infty$, and

    uses the integrable maximal function of II.3.1 and is not asserted here.
  \end{proof}
\end{theorem}

\begin{corollary}[Garnett II.4.2: boundary uniqueness, $H^\infty$ disc case]
  \label{cor:II_4_2}
  \uses{def:boundary_function}
  \notready
  \mainline
  \bookref{Garnett}{II.4.2 ($p=\infty$)}{Printed page 62, PDF page 74}

\begin{definition}[Outer functions on the disc]
  \label{def:outer_factor}
  \lean{outerFunction, IsOuterFunction}
  \leanok
  \uses{def:poisson_integral}
  \bookref{Garnett}{II.4, II.5}{Printed pages 64 and 70, PDF pages 76 and 82}
  For real $k\in L^1(\sigma)$, where $\sigma$ is normalized arclength on
  $\partial\D$, put
  \[
    O_k(z)=\exp\left(\int_{\partial\D}
      \frac{\zeta+z}{\zeta-z}\,k(\zeta)\,d\sigma(\zeta)\right).
  \]
  A function $f$ is outer on $\D$ if $f=C O_k$ on $\D$ for some such $k$
  and a constant $|C|=1$.
\end{definition}

\begin{theorem}[Basic properties of outer functions]
  \label{thm:outer_function_properties}
  \uses{def:outer_factor, def:poisson_integral}
  \lean{outerFunction_ne_zero, analyticOn_outerFunction, norm_outerFunction,
    log_norm_outerFunction, isOuterFunction_outerFunction,
    IsOuterFunction.analyticOn, IsOuterFunction.ne_zero,
    IsOuterFunction.exists_log_norm_eq}
  \leanok
  \bookref{Garnett}{II.4, II.5}{Printed pages 64 and 70, PDF pages 76 and 82}
  For real $k\in L^1(\sigma)$, $O_k$ is analytic and zero-free on $\D$, and
  $\log|O_k|=P[k\,d\sigma]$ on $\D$. Hence every outer function $F$ is
  analytic and zero-free on $\D$, $\log|F|=P[k\,d\sigma]$ for some real
  $k\in L^1(\sigma)$.
  \begin{proof}
    \proves{thm:outer_function_properties}
    \uses{lem:poisson_integral_linear}
    \notready
    The integral $\int\frac{\zeta+z}{\zeta-z}\,k(\zeta)\,d\sigma(\zeta)$ is
    analytic in $z\in\D$, and its real part is $P[k\,d\sigma](z)$ because the
    real part of the Herglotz--Riesz kernel is the Poisson kernel.
    Exponentiating gives the claims for $O_k$; multiplying by $|C|=1$ gives
    those for $F$. No Nevanlinna boundary theorem is used here.
  \end{proof}
\end{theorem}

\begin{theorem}[Garnett II.4.4: prescribed bounded boundary modulus, disc form]
  \label{thm:II_4_4}
  \uses{def:outer_factor, def:boundary_function}
  \notready
  \mainline
  \bookref{Garnett}{II.4.4 ($p=\infty$)}{Printed page 63, PDF page 75}
  Let $h\ge0$ be essentially bounded and not zero almost everywhere on
  $\partial\D$. There is a nonzero $f\in H^\infty(\D)$ with $|f^*|=h$
  almost everywhere if and only if $\log h\in L^1(\sigma)$, where
  $\log0=-\infty$. When this holds, the normalized outer function
  $O_{\log h}$ has the required modulus and satisfies
  $|O_{\log h}(z)|\le\|h\|_\infty$.
  \begin{proof}
    \proves{thm:II_4_4}
    \uses{thm:II_4_1, thm:outer_function_properties, cor:II_3_1(p=∞), thm:I_5_3_ac,
      thm:poisson_reproducing}
    \notready
    Necessity is II.4.1. For sufficiency use Garnett's construction
    $\exp(u+iv)$, where $u=P[\log h\,d\sigma]$ and $v$ is its normalized
    harmonic conjugate. The bound $u\le\log\|h\|_\infty$ gives boundedness.
    Bounded-analytic Fatou gives complex boundary limits, and Poisson Fatou
    identifies their moduli with $h$. The finite-$p$ refinement in the book
    is not needed for the factorization roadmap.
  \end{proof}
\end{theorem}

\begin{theorem}[Boundary modulus of an outer function]
  \label{thm:outer_boundary_modulus}
  \uses{def:outer_factor, def:boundary_function}
  \lean{ae_nontangentiallyConvergentAt_outerFunction,
    ae_norm_boundaryValue_outerFunction}
  \leanok
  \bookref{Garnett}{II.4, outer boundary modulus}{Printed page 64, PDF page 76}
  For real $k\in L^1(\sigma)$, $O_k$ has finite nontangential limits
  almost everywhere and $|O_k^*|=e^k$ almost everywhere.
  \begin{proof}
    \proves{thm:outer_boundary_modulus}
    \uses{thm:II_4_4}
    \notready
    The book's outer construction permits arbitrary integrable logarithmic
    data. To justify finite complex boundary limits without II.5.3, write
    $k=k^+-k^-$ and $O_k=O_{-k^-}/O_{-k^+}$. Apply the bounded construction
    II.4.4 to $e^{-k^-}$ and $e^{-k^+}$; their boundary moduli are positive
    almost everywhere. Taking the quotient gives modulus $e^k$.
    This supplies the disc-form detail behind the boundary assertion after
    the outer definition, independently of Nevanlinna boundary theory.
  \end{proof}
\end{theorem}

\begin{theorem}[Outer uniqueness]
  \label{thm:outer_uniqueness}
  \uses{def:outer_factor, def:boundary_function}
  \lean{IsOuterFunction.exists_eq_const_mul_of_ae_norm_boundaryValue_eq}
  \leanok
  \mainline
  \bookref{Garnett}{II.4, outer uniqueness}{Printed page 64, PDF page 76}
  If two outer functions $F_1,F_2$ satisfy $|F_1^*|=|F_2^*|$ almost
  everywhere on $\partial\D$, then $F_1=cF_2$ on $\D$ for a constant $|c|=1$.
  \begin{proof}
    \proves{thm:outer_uniqueness}
    \uses{thm:outer_boundary_modulus, thm:outer_function_properties}
    \notready
    Write $F_j=C_jO_{k_j}$. By Theorem~\ref{thm:outer_boundary_modulus},
    $e^{k_1}=|F_1^*|=|F_2^*|=e^{k_2}$ almost everywhere, so $k_1=k_2$
    almost everywhere and $\log|F_1|=P[k_1\,d\sigma]=\log|F_2|$ on $\D$. The
    quotient $F_1/F_2$ is analytic with modulus one, hence a unimodular
    constant by the maximum modulus principle.
  \end{proof}
\end{theorem}

\begin{definition}[Nevanlinna class]
  \label{def:Nevanlinna_class}
  \lean{Nevanlinna.MemNevanlinnaDisc,
    Nevanlinna.memNevanlinnaDisc_iff_hasHarmonicMajorant}
  \leanok
  \bookref{Garnett}{II, (5.1)}{Chapter II, Section 5, printed page 66, PDF page 78}
  The Nevanlinna class $N$ consists of analytic functions $f$ on $\D$ such that
  \[
    \sup_{0<r<1}\int_{\partial\D}\log^+|f(r\zeta)|\,d\sigma(\zeta)<\infty,
  \]
  where $\sigma$ is normalized arclength. In Lean, functions are extended by
  zero outside $\D$.

  \emph{Equivalent definition (Garnett (5.1)).} An analytic function $f$ belongs
  to $N$ if and only if $\log^+|f|$ has a harmonic majorant on $\D$.
  \begin{proof}
    \proves{def:Nevanlinna_class}
    \uses{thm:I_6_7, thm:log_abs_subharmonic}
    \leanok
    For the equivalence, apply Theorem~\ref{thm:I_6_7} to the subharmonic
    function $\log^+|f|=\max(\log|f|,0)$. Conversely, a harmonic majorant
    $U\ge\log^+|f|\ge0$ bounds every radial $\log^+$ mean by $U(0)$, by the
    mean-value property. These dependencies concern the equivalence; the
    radial-integral definition itself needs neither result.
  \end{proof}
\end{definition}

\begin{theorem}[Garnett II.5.1: harmonic-majorant characterization of $N$]
  \label{thm:II_5_1}
  \lean{Nevanlinna.memNevanlinnaDisc_iff_exists_signedMeasure_isLeastHarmonicMajorant}
  \leanok
  \mainline
  \bookref{Garnett}{II.5.1}{Chapter II, Section 5, PDF page 78}
  Let $f$ be analytic on $\mathbb{D}$, $f\not\equiv0$. Then $f\in N$ if and
  only if the least harmonic majorant of $\log|f|$ is $P[\mu]$ for a finite
  signed measure $\mu$ on $\partial\mathbb{D}$. Equivalently, $\log|f|$
  admits a majorant $P[\nu]$ for some finite signed boundary measure $\nu$.
  \begin{proof}
    \proves{thm:II_5_1}
    \uses{def:Nevanlinna_class, thm:log_abs_subharmonic, thm:I_6_7, thm:I_3_5(c),
      lem:poisson_integral_linear}
    \leanok
    If $f\in N$, take a nonnegative harmonic majorant $U$ of $\log^+|f|$
    and let $u\le U$ be the least harmonic majorant of $\log|f|$, which
    exists since $f\not\equiv0$. Apply Theorem~\ref{thm:I_3_5(c)} to $U$ and
    $U-u$ and subtract their representing measures to obtain $u=P[\mu]$.
    Conversely, the Jordan identity
    (Lemma~\ref{lem:poisson_integral_linear}) gives, even if $P[\mu]$
    is only a majorant rather than the least one,
    $\log^+|f|\le P[\mu^+]$, a nonnegative harmonic majorant; hence $f\in N$
    by the equivalent definition in Definition~\ref{def:Nevanlinna_class}.
  \end{proof}
\end{theorem}

\begin{lemma}[Garnett II.5.2: remove the zeros of a Nevanlinna function]
  \label{lem:II_5_2}
  \lean{Nevanlinna.MemNevanlinnaDisc.blaschkeCondition,
    Nevanlinna.MemNevanlinnaDisc.exists_eq_blaschkeProduct_mul}
  \leanok
  \mainline
  \bookref{Garnett}{II.5.2}{Chapter II, Section 5, PDF page 78}
  Let $f\in N$ be nonzero and let $B$ be the Blaschke product formed from the
  zeros of $f$, counted with multiplicity. Then $B$ converges and $f=Bg$, where
  the quotient $g=f/B$, with its removable singularities filled in, is analytic
  and zero-free on $\mathbb{D}$, and $g$ still belongs to the Nevanlinna class.
  Moreover, $\log|g|$ is the least harmonic majorant of $\log|f|$ on
  $\mathbb{D}$.
  \begin{proof}
    \proves{lem:II_5_2}
    \uses{thm:II_5_1, thm:II_2_1, thm:II_2_2, thm:II_2_4, def:Nevanlinna_class, thm:I_6_7}
    \notready
    By (5.1), $\log|f|\le\log^+|f|\le U$ for a harmonic $U$, so
    Theorem~\ref{thm:II_2_1} gives the Blaschke condition, and
    Theorem~\ref{thm:I_6_7} gives the least harmonic majorant $u\le U$ of
    $\log|f|$. Since $B$ has the zeros of $f$ with the same multiplicities
    (Theorem~\ref{thm:II_2_2}), $f/B$ has only removable singularities;
    removing them gives a zero-free analytic $g$ with $f=Bg$. As $|B|\le1$,
    $\log|g|$ is a harmonic majorant of $\log|f|$, so $u\le\log|g|$.
    Conversely, $u-\log|g|$ is a harmonic majorant of $\log|B|$, whose least
    harmonic majorant is $0$ by Theorem~\ref{thm:II_2_4}; hence
    $u=\log|g|$. Finally $\log^+|g|=\max(u,0)\le U$, so $g\in N$ by (5.1).
  \end{proof}
\end{lemma}

\begin{theorem}[Garnett II.5.3: boundary values and the least-majorant measure]
  \label{thm:II_5_3}
  \uses{def:nt_limit, def:boundary_function}
  \lean{Nevanlinna.MemNevanlinnaDisc.ae_nontangentiallyConvergentAt,
    Nevanlinna.MemNevanlinnaDisc.ae_boundaryValue_ne_zero,
    Nevanlinna.MemNevanlinnaDisc.integrable_log_norm_boundaryValue,
    Nevanlinna.MemNevanlinnaDisc.exists_mutuallySingular_isLeastHarmonicMajorant}
  \leanok
  \mainline
  \bookref{Garnett}{II.5.3}{Chapter II, Section 5, PDF page 79}
  Let $f\in N$, $f\not\equiv0$. Then:
  \begin{enumerate}
    \item $f$ has a nontangential limit $f^*(\zeta)$ at almost every
      $\zeta\in\partial\D$, and $f^*\ne0$ almost everywhere;
    \item $\log|f^*|\in L^1(d\theta/2\pi)$;
    \item the least harmonic majorant of $\log|f|$ is $P[\mu]$ for a finite
      signed measure $\mu$ on $\partial\D$, unique by
      Theorem~\ref{thm:poisson_measure_unique}, and
      \begin{equation}
        \label{eq:II_5_3_decomposition}
        d\mu=\log|f^*|\,\frac{d\theta}{2\pi}+d\mu_s,
        \qquad \mu_s\perp d\theta .
      \end{equation}
  \end{enumerate}
  In other words, the absolutely continuous part of the representing measure
  is exactly the boundary logarithm, and the whole obstruction to the
  inequality $\log|f(z)|\le\int\log|f^*|P_z\,d\theta/2\pi$ is concentrated in
  the singular part; Garnett's class $N^+$ is the subclass where
  $\mu_s\le0$.
  \begin{proof}
    \proves{thm:II_5_3}
    \uses{thm:II_5_1, lem:II_5_2, thm:II_2_2, thm:I_5_3, cor:II_3_1(p=∞),
      thm:poisson_measure_unique, lem:poisson_integral_linear, cor:II_4_2}
    \notready
    Let $B$ be the Blaschke product over the zeros of $f$ and $g=f/B$. By
    Theorem~\ref{thm:II_5_1} and Lemma~\ref{lem:II_5_2}, $g$ is zero-free and
    $\log|g|=P[\mu]$ is the least harmonic majorant of $\log|f|$; by
    Theorem~\ref{thm:poisson_measure_unique} the measure $\mu$ is unique. Write
    $d\mu=k\,d\theta/2\pi+d\mu_s$ using mathlib's Lebesgue decomposition
    for signed measures. By
    Theorem~\ref{thm:I_5_3}, $\log|g|$ has nontangential limit $k$ almost
    everywhere, and since $|B^*|=1$ almost everywhere
    (Theorem~\ref{thm:II_2_2}) the same holds for $\log|f|$. So $k\in L^1$ and
    \eqref{eq:II_5_3_decomposition} follow once $f$ itself is known to have
    nontangential limits.

    For that, decompose $\log|g|=u_1-u_2$ with $u_j=P[\mu_j]\ge0$ harmonic,
    $\mu=\mu_1-\mu_2$ the Jordan decomposition. Choose harmonic conjugates
    $v_j$ with $v_j(0)=0$, which exist because the disc is simply connected and
    which mathlib provides as
    \texttt{InnerProductSpace.HarmonicOnNhd.exists\_analyticOnNhd\_ball\_re\_eq};
    then $\log g=(u_1+iv_1)-(u_2+iv_2)+ic$ for a real constant $c$, so that
    \[
      g=e^{ic}\,\frac{e^{-(u_2+iv_2)}}{e^{-(u_1+iv_1)}} .
    \]
    Each $e^{-(u_j+iv_j)}$ has modulus $e^{-u_j}\le1$, hence lies in
    $H^\infty$ and has nontangential limits almost everywhere by
    Corollary~\ref{cor:II_3_1(p=∞)}. Each exponential is nonzero on the disc, so II.4.2
    makes its boundary limit nonzero almost everywhere. In particular the
    denominator has a nonzero boundary limit, and the displayed quotient
    gives the finite boundary limits of $g$, and then of $f=Bg$.
    Combining those limits with the finite limit of $\log|g|$ identifies
    $k=\log|f^*|$ and proves (5.2) and (5.3).
  \end{proof}
\end{theorem}

\begin{theorem}[Jordan decomposition of singular measures]
  \label{thm:jordan_decomposition_singular_measure}
  \lean{MeasureTheory.SignedMeasure.totalVariation_apply_eq_zero_iff,
    MeasureTheory.SignedMeasure.mutuallySingular_toENNRealVectorMeasure_iff,
    MeasureTheory.JordanDecomposition.toSignedMeasure_injective}
  \leanok
  \bookref{Garnett}{II, (5.7)}{Printed page 70, PDF page 82}
  Let $\nu$ be a finite real signed measure with Jordan decomposition
  $\nu=\nu^+-\nu^-$. Then $|\nu|(S)=0$ if and only if $\nu^+(S)=\nu^-(S)=0$,
  and $\nu\perp\sigma$ if and only if $\nu^+\perp\sigma$ and $\nu^-\perp\sigma$.
  In particular, if $\nu$ is carried by $\partial\mathbb{D}$ and singular with
  respect to Lebesgue measure, then $\nu^+$ and $\nu^-$ are finite positive
  measures with the same two properties. This decomposition is unique
  among differences of mutually singular positive measures (Jordan uniqueness).
  \begin{proof}
    \proves{thm:jordan_decomposition_singular_measure}
    \leanok
    The total variation is $|\nu|=\nu^++\nu^-$, and a sum of positive measures
    vanishes on $S$, or is singular to $\sigma$, exactly when both summands do.
    Uniqueness is the injectivity of the signed measure associated with a
    Jordan decomposition, supplied by mathlib.
  \end{proof}
\end{theorem}

\begin{definition}[Singular function formula]
  \label{def:singular_function_formula}
  \lean{singularFunction, IsSingularFunction, isSingularFunction_singularFunction}
  \leanok
  \uses{def:poisson_integral}
  \bookref{Garnett}{II, (5.8)}{Printed page 70, PDF page 82}
  For a finite positive measure $\mu$ carried by $\partial\D$ and singular

  \[
    S_\mu(z)=\exp\left(-\int_{\partial\mathbb{D}}
      \frac{\zeta+z}{\zeta-z}\,d\mu(\zeta)\right).
  \]
  A function $S$ is singular on $\D$ if $S=S_\mu$ on $\D$ for some such $\mu$.
\end{definition}

\begin{theorem}[Garnett II, (5.9): modulus of $S_\mu$]
  \label{eq:II_5_9}
  \uses{def:singular_function_formula, def:poisson_integral}
  \lean{singularFunction_ne_zero, analyticOn_singularFunction,
    norm_singularFunction, log_norm_singularFunction, singularFunction_zero,
    IsSingularFunction.analyticOn, IsSingularFunction.ne_zero,
    IsSingularFunction.norm_le_one, IsSingularFunction.zero_pos,
    IsSingularFunction.memHpDisc_indicator}
  \leanok
  \bookref{Garnett}{II, (5.9)}{Printed page 70, PDF page 82}
  For a finite positive measure $\mu$ carried by $\partial\D$, $S_\mu$ is
  analytic and zero-free on $\D$,
  \[
    \log|S_\mu(z)|=-P[\mu](z)\quad(z\in\D),
  \]
  and $S_\mu(0)=e^{-\mu(\partial\D)}$. In particular $|S_\mu|\le1$ on $\D$.
  Every singular function is therefore analytic, zero-free and bounded by one
  on $\D$, has positive value at zero, and belongs to $H^\infty$ after extension
  by zero.
  \begin{proof}
    \proves{eq:II_5_9}
    \uses{lem:poisson_integral_linear}
    \leanok
    The Herglotz--Riesz transform of $\mu$ is analytic on $\D$, and its real
    part is $P[\mu]$ because the real part of the Herglotz--Riesz kernel is the
    Poisson kernel. At $z=0$ the kernel equals $1$ on $\partial\D$.
    Positivity of the Poisson kernel gives $P[\mu]\ge0$, hence $|S_\mu|\le1$.
    The representation defining a singular function transfers these properties
    to $S$; bounded analyticity gives membership in $H^\infty$. No boundary-limit
    theorem or converse characterization is used in these consequences.
  \end{proof}
\end{theorem}

\begin{theorem}[Characterization of singular functions]
  \label{thm:singular_function_characterization}
  \uses{def:singular_function_formula, def:nt_limit}
  \lean{IsSingularFunction.ae_hasNontangentialLimit_norm_eq_one,
    isSingularFunction_iff}
  \leanok
  \bookref{Garnett}{II.5}{Printed page 70, PDF page 82}
  A function $S$ is singular if and only if it is analytic on $\D$,
  has no zeros there, satisfies $|S|\le1$, has nontangential limits of
  modulus one $\sigma$-almost everywhere, and $S(0)$ is positive real. In
  particular every singular function belongs to $H^\infty$.
  \begin{proof}
    \proves{thm:singular_function_characterization}
    \uses{eq:II_5_9, lem:I_5_4, cor:II_3_1(p=∞), thm:I_3_5(c), thm:I_5_3}
    \notready
    The exponential integral is analytic and satisfies
    $\log|S_\mu|=-P[\mu]$. Positivity, the singular Poisson boundary theorem,
    and bounded-analytic Fatou give the four properties.
    Conversely, represent the nonnegative harmonic function $-\log|S|$
    as $P[\mu]$. Its zero boundary limit makes the absolutely continuous
    part of $\mu$ vanish. Then $S/S_\mu$ is a unimodular constant,
    equal to one by positivity at the origin.
  \end{proof}
\end{theorem}

\begin{theorem}[Algebraic structure of $N$]
  \label{thm:nevanlinna_algebra}
  \lean{Nevanlinna.MemNevanlinnaDisc.zero, Nevanlinna.MemNevanlinnaDisc.add,
    Nevanlinna.MemNevanlinnaDisc.smul, Nevanlinna.MemNevanlinnaDisc.mul,

    Nevanlinna.MemNevanlinnaDisc.div, Nevanlinna.memNevanlinnaDisc_of_memHpDisc,
    Nevanlinna.hpDisc_le_nevanlinnaDisc}
  \leanok
  \bookref{Garnett}{II.5}{Chapter II, Section 5, printed page 66, PDF page 78}
  $N$ is closed under sums, scalar multiples, and products. If $f\in N$ has no
  zeros in $\D$, then $1/f\in N$; hence $f/g\in N$ whenever $f,g\in N$ and $g$
  has no zeros in $\D$. Moreover $H^p\subseteq N$ for every $p>0$.
  \begin{proof}
    \proves{thm:nevanlinna_algebra}
    \uses{def:Nevanlinna_class}
    \leanok
    Sums, scalar multiples, and products follow from
    $\log^+|ab|\le\log^+|a|+\log^+|b|$ and
    $\log^+|a+b|\le\log^+|a|+\log^+|b|+\log2$. If $U$ is a harmonic majorant
    of $\log^+|f|$ and $f$ has no zeros, then $U-\log|f|$ is a harmonic
    majorant of $\log^+|1/f|=\log^+|f|-\log|f|$, so $1/f\in N$ by (5.1).
    Finally $H^p\subseteq N$ because $\log^+x\le x^p/p$.
  \end{proof}
\end{theorem}

\begin{theorem}[Outer functions belong to the Nevanlinna class]
  \label{thm:outer_mem_nevanlinna}
  \uses{def:outer_factor}
  \lean{IsOuterFunction.memNevanlinnaDisc_indicator}
  \leanok
  Every outer function on $\D$ belongs to $N$ after extension by zero.
  \begin{proof}
    \proves{thm:outer_mem_nevanlinna}
    \uses{thm:outer_function_properties, def:Nevanlinna_class, thm:harmonic_poisson_integral,
      lem:poisson_integral_linear}
    \leanok
    If $\log|F|=P[k\,d\sigma]$, positivity of the Poisson kernel gives
    $\log^+|F|\le P[k^+\,d\sigma]$, a nonnegative harmonic majorant.
  \end{proof}
\end{theorem}

\section{Garnett II.5.5: canonical factorization}

\begin{definition}[Canonical factorization]
  \label{def:canonical_factorization}
  \lean{Nevanlinna.IsCanonicalFactorization}
  \leanok
  \uses{def:blaschke_condition, def:blaschke_product, def:outer_factor,
      def:singular_function_formula}
  \bookref{Garnett}{II, (5.10)}{Printed page 71, PDF page 83}
  Let $|C|=1$, let $F$ be outer, and let $\mu_1,\mu_2$ be finite positive
  measures carried by $\partial\D$, each singular to normalized arclength,
  with $\mu_1\perp\mu_2$. Then $(C,F,\mu_1,\mu_2)$ is a canonical
  factorization of $f$ if the zeros of $f$ in $\D$, counted with multiplicity,
  satisfy the Blaschke condition and, with $B$ their Blaschke product,
  \[
    f = C\,B\,F\,S_{\mu_1}/S_{\mu_2}\quad\text{on }\D.
  \]
\end{definition}

\begin{theorem}[Garnett II.5.5: existence of canonical factorization in $N$]
  \label{thm:nevanlinna_canonical_factorization}
  \lean{Nevanlinna.MemNevanlinnaDisc.exists_isCanonicalFactorization}
  \leanok
  \uses{def:canonical_factorization, def:outer_factor, def:singular_function_formula}
  \mainline
  \bookref{Garnett}{II.5.5 (existence)}{Chapter II, Section 5, PDF page 83}
  If $f\in N$ and $f\not\equiv0$, then $f$ has a canonical factorization
  $f=C\,B\,F\,S_{\mu_1}/S_{\mu_2}$ on $\D$.
\end{theorem}

\begin{proof}
  \proves{thm:nevanlinna_canonical_factorization}
  \leanok
  \uses{thm:II_5_1, lem:II_5_2, thm:II_5_3, thm:outer_function_properties, eq:II_5_9,
      lem:poisson_integral_linear, thm:jordan_decomposition_singular_measure}
  By Lemma~\ref{lem:II_5_2}, $f=Bg$ with $g\in N$ zero-free and $\log|g|$ the
  least harmonic majorant of $\log|f|$. By (5.2) and (5.3), that majorant is
  $P[k\,d\sigma+\mu_s]$ with $k=\log|f^*|\in L^1(\sigma)$ and
  $\mu_s\perp\sigma$; least harmonic majorants are unique, so
  $\log|g|=P[k\,d\sigma+\mu_s]$ on $\D$. Put $\mu_2=\mu_s^+$ and
  $\mu_1=\mu_s^-$, which are carried by $\partial\D$, singular to $\sigma$,
  and mutually singular. By linearity of the Poisson integral and (5.9),
  $\log|g|=\log|O_k S_{\mu_1}/S_{\mu_2}|$ on $\D$.

  Garnett concludes by taking a single-valued logarithm of the zero-free
  quotient $g$. We instead note that $g/(O_k S_{\mu_1}/S_{\mu_2})$ is analytic
  on $\D$ with modulus one, hence a unimodular constant $C$ by the maximum
  modulus principle. This is equivalent and avoids holomorphic logarithms on
  the disc.
\end{proof}

\begin{theorem}[Garnett II.5.5: uniqueness of canonical factorization]
  \label{thm:nevanlinna_factorization_uniqueness}
  \uses{def:canonical_factorization}
  \lean{Nevanlinna.IsCanonicalFactorization.unique}
  \leanok
  \mainline
  \bookref{Garnett}{II.5.5 (uniqueness)}{Printed page 71, PDF page 83}
  Let $f\in N$ be nonzero. If $(C,F,\mu_1,\mu_2)$ and
  $(C',F',\nu_1,\nu_2)$ are canonical factorizations of $f$, then
  $\mu_1=\nu_1$, $\mu_2=\nu_2$, and $CF=C'F'$ on $\D$.
  \begin{proof}
    \proves{thm:nevanlinna_factorization_uniqueness}
    \uses{thm:II_2_2, thm:II_5_3, thm:outer_uniqueness, thm:outer_boundary_modulus,
      thm:singular_function_characterization, lem:II_5_2, thm:outer_function_properties,
      eq:II_5_9, lem:poisson_integral_linear, thm:poisson_measure_unique,
      thm:jordan_decomposition_singular_measure}
    \leanok
    Follow Garnett. The zeros of $f$ determine $B$. Since
    $|B^*|=|S_1^*|=|S_2^*|=1$ almost everywhere, both outer factors have
    boundary modulus $|f^*|$. Outer uniqueness determines $F$ up to phase;
    in particular $\log|F|=P[\log|f^*|\,d\sigma]$.

    By II.5.2, $u=\log|f/B|$ is the least harmonic majorant of $\log|f|$.
    Formula (5.9) gives
    \[
      u-\log|F|=P[\mu_2-\mu_1]=P[\nu_2-\nu_1].
    \]
    Poisson uniqueness identifies the signed measures, and their mutually
    singular Jordan decompositions identify each positive measure.
    Cancel the common nonzero singular factors and cancel $B$ away from
    its isolated zeros, then extend by continuity, to obtain $CF=C'F'$.

    \emph{Implementation status.} The linked Lean theorem currently has a
    proof using the measure representing $\log|CFS_1/S_2|$ and Lebesgue
    decomposition, without outer boundary uniqueness or II.5.2. That proof
    is preserved, but the book proof displayed here has not yet been
    implemented and is not marked complete.
  \end{proof}
\end{theorem}

\begin{theorem}[Garnett II.5.5: converse]
  \label{thm:nevanlinna_factorization_converse}
  \uses{def:blaschke_product, def:blaschke_condition, def:outer_factor,
      def:singular_function_formula}
  \lean{Nevanlinna.memNevanlinnaDisc_indicator_mul_blaschkeProduct}
  \leanok
  \mainline
  \bookref{Garnett}{II.5.5 (converse)}{Printed page 71, PDF page 83}
  If $B$ is the Blaschke product of a multiplicity satisfying the Blaschke
  condition, $F$ is outer, $S_1,S_2$ are singular, and $C\in\mathbb C$, then
  $CBFS_1/S_2\in N$. The implemented algebraic proof permits arbitrary $C$;
  the book assumes $|C|=1$.
  \begin{proof}
    \proves{thm:nevanlinna_factorization_converse}
    \uses{thm:II_2_2, eq:II_5_9, thm:outer_mem_nevanlinna, thm:nevanlinna_algebra}
    \leanok
    The convergence, analyticity and boundedness parts of II.2.2 give
    $B\in H^\infty\subseteq N$. Its multiplicity and boundary-modulus assertions
    are not used here. The elementary consequences of (5.9) give
    $S_1,S_2\in H^\infty\subseteq N$ and $S_2\ne0$ on $\D$, without using the
    full singular-function characterization. Also $F\in N$ by
    Theorem~\ref{thm:outer_mem_nevanlinna}.

    Closure of $N$ under reciprocals of zero-free functions gives $1/S_2\in N$.
    Multiplication and scalar multiplication then give $CBFS_1/S_2\in N$.
    In Lean each factor is extended by zero outside $\D$; the indicator of
    the product and quotient equals the corresponding product of indicators.

    This is the existing Lean proof. It differs from Garnett's direct signed
    Poisson-majorant argument: the equivalent definition in Definition~\ref{def:Nevanlinna_class} is used upstream for
    outer membership and reciprocal closure, but Theorem~\ref{thm:II_5_1}
    and its signed-measure representation are not invoked.
  \end{proof}
\end{theorem}
