\chapter{Canonical factorization in the Nevanlinna class}

\noindent\textbf{Project overview.}
This project develops a Lean 4 and mathlib formalization of Hardy spaces and
the Nevanlinna class on the unit disc. Its ultimate goal is a formal proof of
the canonical factorization theorem for the Nevanlinna class, decomposing a
nonzero function into its Blaschke, singular, and outer factors. This
blueprint records the mathematical dependency graph and the current
formalization status.

\medskip
\noindent\textbf{Main reference.}
The main mathematical reference is John B. Garnett,
\emph{Bounded Analytic Functions}. Unless another source is stated,
the theorem numbers and chapter/section references below refer to Garnett's
book. The statements in this blueprint are formalization-oriented paraphrases
and are not verbatim reproductions of the source.

\section{Chapter I prerequisites}

\begin{theorem}[Harmonic majorants of subharmonic functions, Garnett I.6.7]
  \label{thm:I_6_7}
  \lean{Subharmonic.hasHarmonicMajorant_iff_radialMean_bddAbove,
    Subharmonic.exists_isLeastHarmonicMajorant_tendsto_poissonModification}
  \leanok
  Source: Garnett, Chapter I, Section 6, PDF page 36.
  Let $v$ be subharmonic on $\mathbb{D}$. Then $v$ has a harmonic majorant on
  $\mathbb{D}$ if and only if
  \[
    \sup_{0<r<1}\frac{1}{2\pi}
      \int_0^{2\pi}v(r e^{i\theta})\,d\theta
    = \sup_{0<r<1} v_r(0) < \infty,
  \]
  where $v_r$ is the Poisson extension to $|z|<r$ of the boundary values of
  $v$ on $|z|=r$. When this condition holds, the least harmonic majorant $u$
  of $v$ is given by
  \[
    u(z)=\lim_{r\to 1}
      \frac{1}{2\pi}\int_0^{2\pi}
        P_{z/r}(\theta)v(r e^{i\theta})\,d\theta
      =\lim_{r\to 1}v_r(z).
  \]
\end{theorem}

\section{Nontangential limits and boundary values}

\begin{definition}[Nontangential approach region]
  \label{def:nt_region}
  \lean{Complex.stolzSetAt}
  \leanok
  Source: Garnett, Chapter I, Section 4, PDF page 23 (disc form).
  For $\zeta\in\partial\D$ and $\alpha>1$ put
  \[
    \Gamma_\alpha(\zeta)
      =\Bigl\{z\in\D:\ |\zeta-z|<\alpha\,(1-|z|)\Bigr\},
  \]
  a teardrop-shaped region asymptotic at $\zeta$ to an angle with vertex
  $\zeta$. Facts to record: $\Gamma_\alpha(\zeta)$ is open and contained in
  $\D$; it increases with $\alpha$; it contains the radial segment
  $[0,1)\zeta$; it is empty when $\alpha\le 1$; and
  $\Gamma_\alpha(1)$ is mathlib's \texttt{Complex.stolzSet}~$\alpha$, with
  $\Gamma_\alpha(\zeta)=\zeta\cdot\Gamma_\alpha(1)$. Mathlib only has the
  region at $\zeta=1$, so the rotated family has to be added.
\end{definition}

\begin{definition}[Nontangential limit]
  \label{def:nt_limit}
  \lean{HasNontangentialLimitWithin, nontangentially, HasNontangentialLimit,
    hasNontangentialLimit_iff_forall}
  \leanok
  \uses{def:nt_region}
  Source: Garnett, Chapter I, Section 5, PDF page 27.
  Write $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$ for the neighbourhood filter
  of $\zeta$ within $\Gamma_\alpha(\zeta)$. The \emph{nontangential filter} at
  $\zeta$ is their supremum in the lattice of filters,
  \[
    \mathcal{N}_\zeta=\sup_{\alpha>1}\ \mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta).
  \]
  For $\alpha>1$, $f$ has limit $L$ at $\zeta$ \emph{within} $\Gamma_\alpha(\zeta)$
  if $f\to L$ along $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$, and $f$ has
  \emph{nontangential limit} $L$ at $\zeta$ if $f\to L$ along
  $\mathcal{N}_\zeta$. The two are related by
  \[
    f\to L \ \text{along}\ \mathcal{N}_\zeta
    \iff
    \forall\,\alpha>1,\ f\to L \ \text{along}\ \mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta).
  \]
\end{definition}

\begin{definition}[Boundary value]
  \label{def:boundary_function}
  \lean{boundaryValue, IsAEBoundaryValue}
  \leanok
  \uses{def:nt_limit}
  The boundary value of $f$ at $\zeta$ is
  \[
    f^*(\zeta)=\lim_{\mathcal{N}_\zeta}f,
  \]
  the limit along the nontangential filter $\mathcal{N}_\zeta$, read through the
  choice operator so that it is a total function. Where the nontangential limit
  at $\zeta$ exists, it equals $f^*(\zeta)$.
\end{definition}

\begin{lemma}[The nontangential filter is nontrivial]
  \label{lem:nt_filter_neBot}
  \lean{radial_mem_stolzSetAt, nontangentially_neBot_of_norm_eq_one,
    ae_nontangentially_neBot_circleMeasure}
  \leanok
  \uses{def:nt_region, def:nt_limit}
  For $\zeta\in\partial\D$ and $\alpha>1$ the radial segment $[0,1)\zeta$ lies in
  $\Gamma_\alpha(\zeta)$; hence $\mathcal{N}_\zeta\neq\bot$.
  \begin{proof}
    \proves{lem:nt_filter_neBot}
    \leanok
    For $0\le r<1$ one has $|r\zeta|=r<1$ and
    $|\zeta-r\zeta|=1-r<\alpha(1-r)$. So $\zeta$ lies in the closure of
    $\Gamma_\alpha(\zeta)$, making $\mathcal{N}_{\Gamma_\alpha(\zeta)}(\zeta)$
    nontrivial; a supremum of filters is nontrivial as soon as one of them is,
    so a single $\alpha$ suffices.
  \end{proof}
\end{lemma}

\begin{lemma}[Boundary values: pointwise, relational and canonical]
  \label{lem:boundary_value_bridge}
  \lean{HasNontangentialLimit.boundaryValue_eq,
    IsAEBoundaryValue.ae_eq_boundaryValue,
    IsAEBoundaryValue.aestronglyMeasurable_boundaryValue}
  \leanok
  \uses{def:boundary_function, lem:nt_filter_neBot}
  Let $\mu$ be a measure on $\partial\D$ carried by the circle.
  \begin{enumerate}
    \item If $f$ has nontangential limit $L$ at $\zeta$, then $f^*(\zeta)=L$.
    \item If $g$ is an almost everywhere nontangential boundary value of $f$,
      then $g=f^*$ almost everywhere.
    \item Consequently $f^*$ is almost everywhere strongly measurable as soon as
      one such $g$ is.
  \end{enumerate}
  Part (3) is why measurability of a boundary function never needs a separate
  argument: it is inherited from whichever explicit representative the relevant
  existence theorem produces.
  \begin{proof}
    \proves{lem:boundary_value_bridge}
    \leanok
    (1) is uniqueness of limits along the nontrivial filter
    $\mathcal{N}_\zeta$; (2) applies (1) at almost every $\zeta$, using
    Lemma~\ref{lem:nt_filter_neBot}; (3) is congruence of almost everywhere
    strong measurability along (2).
  \end{proof}
\end{lemma}

\begin{lemma}[Poisson kernel over a Stolz region]
  \label{lem:poisson_stolz_bound}
  \lean{integral_poissonKernel_circleMeasure_eq_one,
    poissonKernel_le_of_mem_stolzSetAt}
  \leanok
  \uses{def:nt_region}
  Let $\alpha>1$. There is a constant $C_\alpha$ such that for every
  $\zeta\in\partial\D$, every $z\in\Gamma_\alpha(\zeta)$ and every
  $\theta$,
  \[
    P_z(\theta)\ \le\
      \frac{C_\alpha\,(1-|z|)}{\bigl((1-|z|)+|e^{i\theta}-\zeta|\bigr)^{2}} .
  \]
  In particular $P_z(\theta)\le C_\alpha/(1-|z|)$ always, and
  $P_z(\theta)\le C_\alpha (1-|z|)/|e^{i\theta}-\zeta|^{2}$ once
  $|e^{i\theta}-\zeta|\ge 2\alpha(1-|z|)$. Together with
  $\int P_z\,d\theta/2\pi=1$ this exhibits the Poisson kernels, restricted to a
  Stolz region, as an approximate identity with a uniformly integrable
  decreasing majorant. This single estimate replaces Garnett's Theorem I.4.2
  and is the engine of both Lemma~\ref{lem:I_5_4} and
  Theorem~\ref{thm:I_5_3_ac}.
  \begin{proof}
    \proves{lem:poisson_stolz_bound}
    \leanok
    From $|e^{i\theta}-z|\ge 1-|z|$ and, by the triangle inequality and the
    definition of $\Gamma_\alpha(\zeta)$,
    $|e^{i\theta}-z|\ge|e^{i\theta}-\zeta|-\alpha(1-|z|)$, one gets
    $|e^{i\theta}-z|\ge c_\alpha\bigl((1-|z|)+|e^{i\theta}-\zeta|\bigr)$;
    insert this into $P_z(\theta)=(1-|z|^2)/|e^{i\theta}-z|^{2}$ and use
    $1-|z|^2\le 2(1-|z|)$.
  \end{proof}
\end{lemma}

\begin{definition}[Poisson integral of boundary data]
  \label{def:poisson_integral}
  \lean{poissonIntegral, poissonIntegralSigned}
  \leanok
  Boundary data is carried either by a function or by a measure. For a measure
  $\mu$ and data $f$, and for a finite signed measure $\nu$,
  \[
    P[f\,d\mu](w)=\int P_w(z)\,f(z)\,d\mu(z),
    \qquad
    P[\nu](w)=\int P_w(z)\,d\nu(z),
  \]
  the second as an integral against the vector measure $\nu$. The kernel is
  real, so $f$ may take values in any real normed space; $\nu$ is real-valued.
  Taking $f=1$ recovers the Poisson integral of $\mu$ itself.
\end{definition}

\begin{lemma}[The two forms of boundary data agree]
  \label{lem:poisson_integral_withDensity}
  \lean{poissonIntegral_eq_poissonIntegralSigned_withDensityᵥ}
  \leanok
  \uses{def:poisson_integral}
  Let $\mu$ be a finite measure carried by $\partial\D$, let $k\in L^1(\mu)$ and
  let $w$ lie in the open disc. Then
  \[
    P[k\,d\mu](w)=P[k\,\mu](w),
  \]
  where $k\,\mu$ is the signed measure with density $k$ with respect to $\mu$.
  \begin{proof}
    \proves{lem:poisson_integral_withDensity}
    \leanok
    Write $k\,\mu$ through its positive and negative parts, integrate the
    kernel against each, and recombine. The kernel is bounded on
    $\partial\D$ for $w$ interior, so it is integrable against every finite
    measure carried by the circle and against $k$ times such a measure.
  \end{proof}
\end{lemma}

\begin{lemma}[Harnack's inequality on a disc]
  \label{lem:harnack_inequality}
  \lean{InnerProductSpace.harnack_inequality, InnerProductSpace.harnack_le,
    InnerProductSpace.le_harnack, poissonKernel_le_of_mem_ball,
    le_poissonKernel_of_mem_ball}
  \leanok
  Let $u$ be harmonic on a neighbourhood of $\overline{\D}(c,R)$ and nonnegative
  on $\partial\D(c,R)$. Then for $w$ with $\rho=|w-c|<R$,
  \[
    \frac{R-\rho}{R+\rho}\,u(c)\ \le\ u(w)\ \le\ \frac{R+\rho}{R-\rho}\,u(c).
  \]
  The two-sided kernel estimate
  $\frac{R-\rho}{R+\rho}\le P_w(\zeta)\le\frac{R+\rho}{R-\rho}$ underlying it is
  recorded separately, since it is used on its own.
  \begin{proof}
    \proves{lem:harnack_inequality}
    \leanok
    The kernel bounds come from mathlib's estimates on the real part of the
    Herglotz--Riesz kernel. Insert them into the Poisson integral formula
    $u(w)=\int P_w u\,d\sigma$, using $u\ge0$ on the circle to preserve the
    inequalities, and identify $\int u\,d\sigma$ with $u(c)$ by the mean value
    property.
  \end{proof}
\end{lemma}

\begin{theorem}[Poisson integrals of continuous data are harmonic]
  \label{thm:harmonic_poisson_integral}
  \lean{harmonicOnNhd_poissonIntegral, poissonIntegral_eq_circleAverage}
  \leanok
  \uses{def:poisson_integral}
  If $f$ is continuous on $\partial\D(c,R)$ then $P[f\,d\sigma]$ is harmonic on
  the open disc, where $\sigma$ is normalized arclength. For $w$ interior the
  Poisson integral coincides with the circle average of $P_w\cdot f$.
  \begin{proof}
    \proves{thm:harmonic_poisson_integral}
    \leanok
    Harmonicity of the circle average follows from harmonicity of the kernel in
    the interior variable, by differentiating under the integral sign; the
    identification with the Poisson integral is the circle average / circle
    measure transport, whose integrability hypothesis holds because both
    factors are continuous on the circle.
  \end{proof}
\end{theorem}

\begin{theorem}[Differentiation of measures on the circle]
  \label{thm:circle_differentiation}
  \lean{MeasureTheory.Besicovitch.ae_tendsto_rnDeriv,
    MeasureTheory.VitaliFamily.ae_eventually_measure_zero_of_singular,
    MeasureTheory.Besicovitch.tendsto_filterAt,
    ae_tendsto_average_abs_sub_circleMeasure,
    norm_circleMap_sub_circleMap, abs_sub_le_norm_circleMap_sub,
    circleMeasure_closedBall_le_of_norm_eq_one}
  \leanok
  Put $\sigma=\texttt{circleMeasure }0\ 1$. If $\nu$ is a finite positive
  Borel measure on $\C$, then for $\sigma$-almost every $\zeta$,
  \[
    \lim_{h\downarrow0}
      \frac{\nu\bigl(\overline B(\zeta,h)\bigr)}
           {\sigma\bigl(\overline B(\zeta,h)\bigr)}
      =\frac{d\nu}{d\sigma}(\zeta),
  \]
  and in particular the quotient tends to zero almost everywhere when
  $\nu\perp\sigma$. Moreover, almost every $\zeta$ is a Lebesgue point of a
  given $k\in L^1(\sigma)$:
  \[
    \lim_{h\downarrow0}\ \frac{1}{\sigma(\overline B(\zeta,h))}
      \int_{\overline B(\zeta,h)}|k-k(\zeta)|\,d\sigma\ =\ 0.
  \]
  For $\|\zeta\|=1$ the supporting chord--arc estimate also gives
  $\sigma(\overline B(\zeta,h))\le \operatorname{ofReal}(h/2)$.
  \begin{proof}
    \proves{thm:circle_differentiation}
    \leanok
    Apply mathlib's Besicovitch Vitali family directly on $\C$ and transport
    its filter to shrinking closed balls. Its singular-measure theorem gives
    the zero-density assertion, and its differentiation theorem gives the
    Radon--Nikodym derivative and Lebesgue points. The chord--arc estimate is
    Jordan's sine inequality applied to the circle parametrization.
  \end{proof}
\end{theorem}

\begin{lemma}[Uniform ball-density bound for positive Poisson integrals]
  \label{lem:poisson_ball_density_bound}
  \lean{integral_poissonKernel_le_of_measure_closedBall_le}
  \leanok
  \uses{def:poisson_integral, lem:poisson_stolz_bound,
    thm:circle_differentiation}
  Let $\nu$ be a finite positive measure carried by $\partial\D$, let
  $\|\zeta\|=1$, and suppose that for every $r>0$,
  \[
    \nu(\overline B(\zeta,r))
      \le c\,\sigma(\overline B(\zeta,r)).
  \]
  Then for $z\in\Gamma_\alpha(\zeta)$,
  \[
    \int_{\partial\D}P_z\,d\nu\le 8(\alpha+2)^2c.
  \]
  \begin{proof}
    \proves{lem:poisson_ball_density_bound}
    \leanok
    Decompose the circle into dyadic chordal annuli centred at $\zeta$: with
    $\delta=1-|z|$, put $S_0=\overline B(\zeta,\delta)$ and
    $S_j=\overline B(\zeta,2^j\delta)\setminus\overline B(\zeta,2^{j-1}\delta)$.
    On $S_j$ the Stolz-region kernel bound gives
    $P_z\le 4A/(4^j\delta)$ with $A=2(\alpha+2)^2$, while the density
    hypothesis and the chord--arc estimate give
    $\nu(S_j)\le c\,2^j\delta/2$; the product is $2Ac\,2^{-j}$ and
    $\sum_j 2^{-j}=2$, so the total is $4Ac=8(\alpha+2)^2c$, exactly the
    stated constant.

    In Lean the estimate is carried out for the lower integral, so that no
    piecewise integrability is needed, and transferred at the end by
    \texttt{integral\_eq\_lintegral\_of\_nonneg\_ae}. Using
    \texttt{lintegral\_iUnion\_le} rather than \texttt{lintegral\_iUnion}
    means only that the annuli cover the plane has to be proved, not that they
    are pairwise disjoint.
  \end{proof}
\end{lemma}

\begin{lemma}[Density-zero Poisson integrals]
  \label{lem:poisson_density_zero}
  \lean{hasNontangentialLimit_integral_poissonKernel_of_density_zero}
  \leanok
  \uses{def:nt_limit, def:poisson_integral,
    lem:poisson_ball_density_bound, lem:poisson_stolz_bound}
  Let $\nu$ be a finite positive measure carried by $\partial\D$. If
  \[
    \frac{\nu(\overline B(\zeta,r))}
         {\sigma(\overline B(\zeta,r))}\longrightarrow0
    \qquad(r\downarrow0),
  \]
  then $P[\nu](z)\to0$ nontangentially as $z\to\zeta$.
  \begin{proof}
    \proves{lem:poisson_density_zero}
    \leanok
    Restrict $\nu$ to a small fixed ball about $\zeta$ and its complement.
    Density zero supplies a uniform ball-density bound for the near part, so
    Lemma~\ref{lem:poisson_ball_density_bound} makes its Poisson integral
    arbitrarily small. The far part stays a fixed positive distance from
    $\zeta$, and its Poisson integral tends uniformly to zero in every Stolz
    region.
  \end{proof}
\end{lemma}

\begin{lemma}[Garnett I.5.4: singular Poisson integrals vanish at the boundary]
  \label{lem:I_5_4}
  \lean{isAEBoundaryValue_poissonIntegralSigned_zero_of_mutuallySingular}
  \leanok
  \uses{def:nt_limit, def:poisson_integral, lem:poisson_density_zero,
    thm:circle_differentiation}
  Source: Garnett, Chapter I, Section 5, PDF pages 41--42.
  If $\nu$ is a finite signed measure on $\partial\D$ that is singular with
  respect to $d\theta$, then
  \[
    \operatorname{nt}\lim_{z\to\zeta}P[\nu](z)=0
  \]
  for almost every $\zeta\in\partial\D$. This is the step that removes the
  singular part in Theorem~\ref{thm:I_5_3}, and it is also what gives the
  singular functions of Definition~\ref{def:singular_function_formula} boundary
  modulus one almost everywhere.
  \begin{proof}
    \proves{lem:I_5_4}
    \leanok
    Mutual singularity makes the total variation of $\nu$ singular with
    respect to $\sigma$. Expanding the Jordan decomposition and using
    subadditivity of vector-measure variation gives
    $\operatorname{var}(\nu)\le |\nu|$, so the variation measure is also
    carried by the circle and singular to $\sigma$. Theorem~\ref{thm:circle_differentiation}
    gives density zero almost everywhere, and Lemma~\ref{lem:poisson_density_zero}
    gives $P[\operatorname{var}(\nu)]\to0$. Finally
    $|P[\nu]|\le P[\operatorname{var}(\nu)]$ follows from the vector-integral
    norm inequality, and squeezing proves the claim.
  \end{proof}
\end{lemma}

\begin{theorem}[Fatou at Lebesgue points]
  \label{thm:I_5_3_ac}
  \lean{hasNontangentialLimit_poissonIntegral_of_lebesguePoint,
    isAEBoundaryValue_poissonIntegral}
  \leanok
  \uses{def:nt_limit, def:poisson_integral, lem:poisson_density_zero,
    thm:circle_differentiation}
  Let $k\in L^1(d\theta/2\pi)$. Then at every Lebesgue point $\zeta$ of $k$,
  and hence almost everywhere,
  \[
    \operatorname{nt}\lim_{z\to\zeta}P\bigl[k\,d\theta/2\pi\bigr](z)=k(\zeta).
  \]
  \begin{proof}
    \proves{thm:I_5_3_ac}
    \leanok
    At a Lebesgue point put
    $d\rho_\zeta=|k-k(\zeta)|\,d\sigma$. Its ball-density quotient is exactly
    the mean oscillation of $k$, hence tends to zero. Apply
    Lemma~\ref{lem:poisson_density_zero} to $\rho_\zeta$, use
    $\int P_z\,d\sigma=1$, and bound
    $|P[k](z)-k(\zeta)|$ by $P[\rho_\zeta](z)$. The almost-everywhere statement
    follows from Theorem~\ref{thm:circle_differentiation}.
  \end{proof}
\end{theorem}

\begin{theorem}[Garnett I.5.3: Fatou theorem, $p=1$ disc form]
  \label{thm:I_5_3}
  \lean{isAEBoundaryValue_poissonIntegralSigned_rnDeriv,
    boundaryValue_poissonIntegralSigned, integrable_boundaryValue_poissonIntegralSigned}
  \leanok
  \uses{thm:I_5_3_ac, lem:I_5_4,
    def:nt_limit, def:boundary_function, def:poisson_integral,
    lem:poisson_integral_withDensity, lem:boundary_value_bridge}
  Source: Garnett, Chapter I, Section 5, PDF pages 40--41.
  Let $\mu$ be a finite signed measure on $\partial\D$, with Lebesgue
  decomposition $d\mu=k\,d\theta/2\pi+d\mu_s$, $\mu_s\perp d\theta$. Then the
  Poisson integral of $\mu$ has a nontangential limit at almost every boundary
  point, and
  \[
    \operatorname{nt}\lim_{z\to\zeta}P[\mu](z)=k(\zeta)
    \qquad\text{for almost every }\zeta\in\partial\D;
  \]
  equivalently $\bigl(P[\mu]\bigr)^*=k$ almost everywhere, so the boundary
  function is in $L^1$. Garnett states the stronger half-plane theorem for all
  $1\le p\le\infty$; only this $p=1$ disc form is used in this project.
  \begin{proof}
    \proves{thm:I_5_3}
    \leanok
    Split $\mu$ by the Lebesgue decomposition, which mathlib provides as
    \texttt{MeasureTheory.SignedMeasure.singularPart\_add\_withDensity\_rnDeriv\_eq}, and add
    Theorem~\ref{thm:I_5_3_ac} to Lemma~\ref{lem:I_5_4}, using that
    nontangential limits are limits along a filter and therefore additive.
    The density part is identified with its signed Poisson integral by
    Lemma~\ref{lem:poisson_integral_withDensity}; integrability of the kernel
    against both summands then identifies their sum with $P[\mu]$. Finally use
    Lemma~\ref{lem:boundary_value_bridge} for the canonical boundary function.
  \end{proof}
\end{theorem}

\begin{theorem}[Bounded harmonic functions are Poisson integrals of $L^\infty$ data]
  \label{thm:bounded_harmonic_poisson_Linfty}
  \notready
  \uses{thm:positive_harmonic_poisson_measure}
  If $u$ is real harmonic on $\D$ with $|u|\le M$, then there is
  $k\in L^\infty(d\theta/2\pi)$ with $\|k\|_\infty\le M$ and $u=P[k\,
  d\theta/2\pi]$.
  \begin{proof}
    \notready
    Apply Theorem~\ref{thm:positive_harmonic_poisson_measure} to the positive
    harmonic function $u+M$ to get $u+M=P[\mu]$ with $\mu\ge0$ finite. Since
    $\mu$ is the weak-star limit of the measures $(u(r\,\cdot)+M)\,
    d\theta/2\pi$, whose densities are bounded by $2M$, one gets
    $\mu\le 2M\,d\theta/2\pi$; Radon--Nikodym then produces a density bounded
    by $2M$, and subtracting $M$ gives $k$.
  \end{proof}
\end{theorem}

\begin{theorem}[A boundary measure is determined by its Poisson integral]
  \label{thm:poisson_measure_unique}
  \notready
  If $\mu$ and $\nu$ are finite signed measures on $\partial\D$ with
  $P[\mu]=P[\nu]$ on $\D$, then $\mu=\nu$. The proof is the standard weak-star
  argument: $P[\mu](r\,\cdot)\,d\theta/2\pi\to\mu$ when tested against
  continuous functions on the circle. This is what licenses speaking of
  \emph{the} representing measure in Theorem~\ref{thm:II_5_3}, and it is used
  again for the uniqueness of the singular factors
  (Theorem~\ref{thm:singular_measure_determines_factors}) and in Garnett's
  Theorem II.5.4.
\end{theorem}

\section{Chapter II, Section 2: Blaschke products}

\begin{theorem}[Garnett II.2.1: harmonic majorants imply the Blaschke condition]
  \label{thm:II_2_1}
  \lean{summable_one_sub_norm_of_hasHarmonicMajorant,
    tsum_one_sub_norm_le_of_isLeastHarmonicMajorant}
  \leanok
  \uses{thm:I_6_7}
  Source: Garnett, Chapter II, Section 2, PDF page 63.
  Let $f$ be a nonzero analytic function on $\mathbb{D}$, and let
  $\{a_n\}$ be its zeros, counted with multiplicity. If $\log |f|$ has a
  harmonic majorant on $\mathbb{D}$, then
  \[
    \sum_n (1-|a_n|)<\infty.
  \]
  Moreover, if $f(0)\ne 0$ and $u$ is the least harmonic majorant of
  $\log |f|$, then
  \[
    \sum_n (1-|a_n|) \le u(0)-\log |f(0)|.
  \]
  By Theorem~\ref{thm:II_5_1}, the harmonic-majorant hypothesis is equivalent
  to $f$ belonging to the Nevanlinna class $N$.
  \begin{proof}
    After removing a possible zero at the origin, fix $r<1$ avoiding the
    moduli of the zeros and divide $f(rz)$ by the finite Blaschke product with
    the same zeros. The logarithm of the modulus of the resulting zero-free
    analytic function is harmonic, so the mean-value formula gives the
    familiar Jensen identity
    \[
      \log |f(0)|+\sum_{|a_j|<r}\log\frac{r}{|a_j|}
      =\frac{1}{2\pi}\int_0^{2\pi}\log|f(r e^{i\theta})|\,d\theta.
    \]
    Theorem~\ref{thm:I_6_7} identifies the limit of the right-hand side with
    $u(0)$. Letting $r\to1$ and using
    $1-|a_j|\le\log(1/|a_j|)$ proves both conclusions.
  \end{proof}
\end{theorem}

\begin{theorem}[Garnett II.2.2: convergence and boundary size of Blaschke products]
  \label{thm:II_2_2}
  \notready
  Source: Garnett, Chapter II, Section 2, PDF page 64.
  If $\{a_n\}\subset\mathbb{D}$ satisfies the Blaschke condition, then the
  associated Blaschke product converges uniformly on compact subsets of
  $\mathbb{D}$ to a bounded analytic function whose zeros are exactly the
  $a_n$ with the prescribed multiplicities, and whose boundary modulus is one
  almost everywhere.
\end{theorem}

\begin{theorem}[Garnett II.2.3 (F. Riesz): division by the Blaschke factor]
  \label{thm:II_2_3}
  \notready
  \uses{thm:II_2_1, thm:II_2_2}
  Source: Garnett, Chapter II, Section 2, PDF page 66.
  Let $0<p<\infty$, let $f\in H^p(\mathbb D)$ be nonzero, and let $B$ be the
  Blaschke product formed from all zeros of $f$, counted with multiplicity.
  Then $g=f/B$ extends analytically across the removable singularities,
  belongs to $H^p(\mathbb D)$, and satisfies
  \[
    \lVert g\rVert_{H^p}=\lVert f\rVert_{H^p}.
  \]
\end{theorem}

\begin{theorem}[Garnett II.2.4: characterization of Blaschke products]
  \label{thm:II_2_4}
  \notready
  \uses{thm:I_6_7, thm:II_2_1, thm:II_2_2}
  Source: Garnett, Chapter II, Section 2, PDF page 66.
  Let $f\in H^\infty(\mathbb D)$ with $\lVert f\rVert_\infty\le 1$. Then the
  following are equivalent:
  \begin{enumerate}
    \item $f=\lambda B$, where $|\lambda|=1$ and $B$ is a Blaschke product;
    \item
      \[
        \lim_{r\to1}\frac{1}{2\pi}\int_0^{2\pi}
          \log|f(r e^{i\theta})|\,d\theta=0;
      \]
    \item the least harmonic majorant of $\log|f|$ is the zero function.
  \end{enumerate}
  The equivalence of the last two conditions is the direct use of
  Theorem~\ref{thm:I_6_7}; condition (3) is the form used in
  Lemma~\ref{lem:II_5_2}.
\end{theorem}

\section{Chapter II, Section 4: boundary uniqueness and outer functions}

\begin{corollary}[Garnett II.4.2: boundary uniqueness]
  \label{cor:II_4_2}
  \notready
  \uses{thm:II_5_3, def:boundary_function}
  Source: Garnett, Chapter II, Section 4, PDF page 75.
  Let $f\in N$, and suppose the boundary function $f^*$ vanishes on a subset
  of $\partial\D$ of positive Lebesgue measure. Then $f\equiv0$. Since
  $H^p\subseteq N$ for every $p>0$, this contains Garnett's statement for
  $H^p(\D)$, $0<p\le\infty$.
  \begin{proof}
    \notready
    Contrapositive. If $f\not\equiv0$ then part (2) of
    Theorem~\ref{thm:II_5_3} gives $\log|f^*|\in L^1(d\theta/2\pi)$, so
    $\log|f^*|>-\infty$ and hence $f^*\ne0$ almost everywhere. The passage
    from $H^p$ to $N$ is the inclusion \texttt{hpDisc\_le\_nevanlinnaDisc},
    already formalized.

    Garnett argues the other way round, deducing this from Theorem II.4.1,
    whose proof dominates $\log^+|f(re^{i\theta})|$ by $(1/p)|f^*|^p$ with
    $f^*$ the nontangential maximal function of Theorem II.3.1. Reversing the
    implication keeps the present development free of the maximal function and
    of the Hardy--Littlewood theory behind it, at the cost of making this
    corollary depend on Theorem~\ref{thm:II_5_3}. There is no circularity:
    Theorem~\ref{thm:II_5_3} obtains the nonvanishing of the boundary values
    of $e^{-(u_j+iv_j)}$ from a second application of
    Theorem~\ref{thm:I_5_3}, not from this corollary. What is lost is
    Garnett's sharper inequality~II.4.1 itself,
    $\log|f(z_0)|\le\int\log|f^*|P_{z_0}\,d\theta/2\pi$ for $f\in H^p$, which
    is not needed for Theorem~\ref{thm:nevanlinna_canonical_factorization}; it
    would be needed for the $N^+$ refinements, and there the maximal function
    would have to be developed anyway.
  \end{proof}
\end{corollary}

\begin{theorem}[Garnett II.4.4: construction from logarithmic data]
  \label{thm:II_4_4}
  \notready
  Source: Garnett, Chapter II, Section 4, PDF page 75.
  Given a nonnegative boundary function $\varphi$ with
  $\log\varphi\in L^1(\partial\mathbb{D})$, there is an outer analytic
  function $F$ on $\mathbb{D}$ whose a.e. boundary modulus is $\varphi$.
  Equivalently, $F$ is obtained by exponentiating the analytic function whose
  real part is the Poisson integral of $\log\varphi$.
\end{theorem}

\begin{theorem}[Outer uniqueness]
  \label{thm:outer_uniqueness}
  \notready
  \uses{cor:II_4_2, def:boundary_function}
  If two outer functions have the same boundary modulus almost everywhere,
  then their quotient is a unimodular constant. Equivalently, the boundary
  modulus determines the outer factor up to the expected constant.
\end{theorem}

\begin{definition}[Outer factor]
  \label{def:outer_factor}
  \notready
  \uses{thm:II_4_4, thm:outer_uniqueness}
  The outer factor $F$ in the Nevanlinna factorization is the zero-free
  analytic factor built from the absolutely continuous part of the boundary
  measure representing the least harmonic majorant.
\end{definition}

\begin{theorem}[Garnett II.5.1: harmonic-majorant characterization of $N$]
  \label{thm:II_5_1}
  \notready
  \uses{thm:log_abs_subharmonic, thm:I_6_7,

  Source: Garnett, Chapter II, Section 5, PDF page 78.
  For an analytic function $f$ on $\mathbb{D}$, the following are equivalent:
  $f$ belongs to the Nevanlinna class $N$, and the subharmonic function

\end{theorem}

\begin{theorem}[Logarithm of modulus is subharmonic]

  \begin{proof}
    \proves{thm:log_abs_subharmonic}
    \leanok

  \end{proof}

\begin{theorem}[Garnett I.3.5(c): positive harmonic functions as Poisson integrals, disc form]
  \label{thm:positive_harmonic_poisson_measure}
  \lean{harmonicOnNhd_nonneg_iff_exists_eq_integral_poissonKernel,
    harmonicOnNhd_integral_poissonKernel, integral_poissonKernel_nonneg}
  \leanok
  Source: Garnett, Chapter I, Section 3, Theorem 3.5(c), especially the
  disc-version argument in its proof, printed pages 17--19 (PDF pages
  approximately 29--31).
  If $u$ is a positive harmonic function on $\mathbb{D}$, then there is a
  finite positive measure $\mu$ on $\partial\mathbb{D}$ such that
  $u=P[\mu]$. Conversely, the Poisson integral of any finite positive boundary
  measure is positive and harmonic.

  Formalization status: the Lean statements are in place, the boundary measure
  being modelled as a finite measure on $\mathbb{C}$ carried by the circle. The
  converse direction is proved; the analyticity of the Herglotz--Riesz
  transform of a boundary measure, which is the engine of that direction, is
  available as \texttt{analyticOnNhd\_integral\_herglotzRieszKernel}. The existence
  direction, whose proof needs weak-$*$ compactness of the approximating
  measures $u(r\,\cdot)\,d\sigma$, is still open.
\end{theorem}

\begin{lemma}[Garnett II.5.2: remove the zeros of a Nevanlinna function]
  \label{lem:II_5_2}
  \notready
  \uses{thm:II_5_1, thm:II_2_1, thm:II_2_2, thm:II_2_4}
  Source: Garnett, Chapter II, Section 5, PDF page 78.
  Let $f\in N$ be nonzero and let $B$ be the Blaschke product formed from the
  zeros of $f$, counted with multiplicity. Then $B$ converges, the quotient
  $g=f/B$ is analytic and zero-free on $\mathbb{D}$, and $g$ still belongs to
  the Nevanlinna class. Moreover, $\log|g|$ is the least harmonic majorant of
  $\log|f|$ on $\mathbb{D}$.
\end{lemma}

\begin{corollary}[$H^\infty$ boundary values, disc form]
  \label{cor:II_3_2}
  \notready
  \uses{thm:bounded_harmonic_poisson_Linfty, thm:I_5_3,
    def:boundary_function}
  Source: this is not one of Garnett's numbered statements. It is the step
  taken from the proof of Garnett's Corollary II.3.2 (Chapter II, Section 3,
  PDF page 68) that reads ``suppose $f(z)$ is some $H^p$ function; if $p>1$,
  then by Theorem I.5.3, $f(z)$ has nontangential limit $f(t)$ and
  $f(z)=P_y*f(x)$'', specialized to $p=\infty$ on the disc, together with the
  norm bound, which Garnett does not state.
  Every $f\in H^\infty(\D)$ has a nontangential limit at almost every boundary
  point, and its boundary function satisfies
  $\|f^*\|_{L^\infty(d\theta/2\pi)}\le\|f\|_\infty$.
  \begin{proof}
    \notready
    The real and imaginary parts of $f$ are bounded harmonic, hence Poisson
    integrals of $L^\infty$ boundary data by
    Theorem~\ref{thm:bounded_harmonic_poisson_Linfty}; an $L^\infty$ density is
    in particular an $L^1$ density, so Theorem~\ref{thm:I_5_3} gives
    nontangential limits almost everywhere, and identifies $f^*$ almost
    everywhere with that $L^\infty$ density, which is what makes $f^*$
    measurable. The norm bound needs no maximal function:
    $\|f(z)\|\le\|f\|_\infty$ pointwise on $\D$, and the bound passes to the
    limit along the filter. Garnett's Corollary II.3.2 itself is the
    characterization of which $L^p$ functions are boundary values of $H^p$
    functions; that characterization and the remaining $H^p$ cases are not
    needed here.
  \end{proof}
\end{corollary}

\begin{theorem}[Garnett II.5.3: boundary values and the least-majorant measure]
  \label{thm:II_5_3}
  \notready
  \uses{thm:II_5_1, lem:II_5_2, thm:II_2_2, thm:I_5_3, cor:II_3_2,
    thm:poisson_measure_unique, def:nt_limit, def:boundary_function}
  Source: Garnett, Chapter II, Section 5, PDF page 79.
  Let $f\in N$, $f\not\equiv0$. Then:
  \begin{enumerate}
    \item $f$ has a nontangential limit $f^*(\zeta)$ at almost every
      $\zeta\in\partial\D$, and $f^*\ne0$ almost everywhere;
    \item $\log|f^*|\in L^1(d\theta/2\pi)$;
    \item the least harmonic majorant of $\log|f|$ is $P[\mu]$ for a unique
      finite signed measure $\mu$ on $\partial\D$, and
      \begin{equation}
        \label{eq:II_5_3_decomposition}
        d\mu=\log|f^*|\,\frac{d\theta}{2\pi}+d\mu_s,
        \qquad \mu_s\perp d\theta .
      \end{equation}
  \end{enumerate}
  In other words, the absolutely continuous part of the representing measure
  is exactly the boundary logarithm, and the whole obstruction to the
  inequality $\log|f(z)|\le\int\log|f^*|P_z\,d\theta/2\pi$ is concentrated in
  the singular part; Garnett's class $N^+$ is the subclass where
  $\mu_s\le0$.
  \begin{proof}
    \notready
    Let $B$ be the Blaschke product over the zeros of $f$ and $g=f/B$. By
    Theorem~\ref{thm:II_5_1} and Lemma~\ref{lem:II_5_2}, $g$ is zero-free and
    $\log|g|=P[\mu]$ is the least harmonic majorant of $\log|f|$; by
    Theorem~\ref{thm:poisson_measure_unique} the measure $\mu$ is unique. Write
    $d\mu=k\,d\theta/2\pi+d\mu_s$ using mathlib's Lebesgue decomposition
    for signed measures. By
    Theorem~\ref{thm:I_5_3}, $\log|g|$ has nontangential limit $k$ almost
    everywhere, and since $|B^*|=1$ almost everywhere
    (Theorem~\ref{thm:II_2_2}) the same holds for $\log|f|$. So $k\in L^1$ and
    \eqref{eq:II_5_3_decomposition} follow once $f$ itself is known to have
    nontangential limits.

    For that, decompose $\log|g|=u_1-u_2$ with $u_j=P[\mu_j]\ge0$ harmonic,
    $\mu=\mu_1-\mu_2$ the Jordan decomposition. Choose harmonic conjugates
    $v_j$ with $v_j(0)=0$, which exist because the disc is simply connected and
    which mathlib provides as
    \texttt{InnerProductSpace.HarmonicOnNhd.exists\_analyticOnNhd\_ball\_re\_eq};
    then $\log g=(u_1+iv_1)-(u_2+iv_2)+ic$ for a real constant $c$, so that
    \[
      g=e^{ic}\,\frac{e^{-(u_2+iv_2)}}{e^{-(u_1+iv_1)}} .
    \]
    Each $e^{-(u_j+iv_j)}$ has modulus $e^{-u_j}\le1$, hence lies in
    $H^\infty$ and has nontangential limits almost everywhere by
    Corollary~\ref{cor:II_3_2}. Those limits are nonzero almost everywhere:
    $\log|e^{-(u_j+iv_j)}|=-P[\mu_j]$ has a finite nontangential limit almost
    everywhere by Theorem~\ref{thm:I_5_3} again, so the boundary modulus is
    $e^{-k_j}>0$ almost everywhere. Hence $g$, and then $f=Bg$, have
    nontangential limits almost everywhere.

    Garnett obtains the nonvanishing from the boundary uniqueness theorem
    (Corollary~\ref{cor:II_4_2}) instead. Using
    Theorem~\ref{thm:I_5_3} as above keeps II.5.3 independent of the $H^p$
    boundary uniqueness theory, which is why \texttt{cor:II\_4\_2} is not among
    the dependencies above; it is still needed for the outer factor.
  \end{proof}
\end{theorem}

\section{Singular factors}

\begin{theorem}[Jordan decomposition of singular measures]
  \label{thm:jordan_decomposition_singular_measure}
  \notready
  If $\nu$ is a finite real signed measure on $\partial\mathbb{D}$ and is
  singular with respect to Lebesgue measure, then its Jordan decomposition
  $\nu=\nu^+-\nu^-$ has $\nu^+$ and $\nu^-$ finite positive singular measures.
\end{theorem}

\begin{definition}[Singular function formula]
  \label{def:singular_function_formula}
  \notready
  For a finite positive singular measure $\sigma$ on $\partial\mathbb{D}$, the
  associated singular function is the analytic exponential
  \[
    S_\sigma(z)=\exp\left(-\int_{\partial\mathbb{D}}
      \frac{\zeta+z}{\zeta-z}\,d\sigma(\zeta)\right).
  \]
  It is bounded by one in the disc and has boundary modulus one almost
  everywhere.
\end{definition}

\begin{definition}[Singular factors]
  \label{def:singular_factors}
  \notready
  \uses{thm:jordan_decomposition_singular_measure,
    def:singular_function_formula, lem:I_5_4}
  If the singular part of the representing measure is
  $\nu_s=\nu_s^+-\nu_s^-$, then the canonical factorization uses the two
  singular functions $S_1=S_{\nu_s^+}$ and $S_2=S_{\nu_s^-}$.
\end{definition}

\section{Final factorization and uniqueness}

\begin{theorem}[Zeros determine the Blaschke factor]
  \label{thm:zeros_determine_blaschke}
  \notready
  The multiset of zeros in $\mathbb{D}$ determines the associated Blaschke
  product up to the chosen unimodular normalization. With a fixed convention
  for elementary factors, the product is uniquely determined.
\end{theorem}

\begin{theorem}[Boundary modulus determines the outer factor]
  \label{thm:boundary_modulus_determines_outer}
  \notready
  \uses{thm:outer_uniqueness}
  If two outer functions have the same a.e. boundary modulus, then they differ
  by multiplication by a constant of modulus one.
\end{theorem}

\begin{theorem}[Singular measure determines singular factors]
  \label{thm:singular_measure_determines_factors}
  \notready
  \uses{def:singular_factors}
  The positive and negative singular parts of the representing measure
  determine $S_1$ and $S_2$. Conversely, equality of the corresponding
  singular functions forces equality of the singular measures, modulo the
  normalization convention.
\end{theorem}

\begin{theorem}[Uniqueness of Nevanlinna factorization]
  \label{thm:nevanlinna_factorization_uniqueness}
  \notready
  \uses{thm:zeros_determine_blaschke,
    thm:boundary_modulus_determines_outer,
    thm:singular_measure_determines_factors}
  In the canonical factorization, the Blaschke factor is determined by the
  zero multiset, the outer factor by the a.e. boundary modulus, and the
  singular factors by the singular part of the representing measure. The only
  remaining ambiguity is the conventional unimodular constant.
\end{theorem}

\begin{theorem}[Garnett II.5.5: canonical factorization in $N$]
  \label{thm:nevanlinna_canonical_factorization}
  \notready
  \uses{thm:II_5_1, lem:II_5_2, thm:II_5_3, def:outer_factor,
    def:singular_factors, thm:nevanlinna_factorization_uniqueness}
  Source: Garnett, Chapter II, Section 5, PDF page 83.
  If $f\in N$ and $f$ is not identically zero, then there exist a unimodular
  constant $C$, a Blaschke product $B$, an outer function $F$, and singular
  functions $S_1,S_2$ such that
  \[
    f = C\,B\,F\,S_1/S_2.
  \]
  Conversely, every expression of this form belongs to the Nevanlinna class.
  The factors are unique after fixing the standard normalizations.
\end{theorem}

\begin{proof}
  \notready
  Remove the zeros using the Blaschke product. Apply the boundary-value and
  logarithmic-integrability theorem to the zero-free quotient. Split the
  representing measure of the least harmonic majorant into absolutely
  continuous and singular parts. The absolutely continuous part gives the
  outer factor, while the positive and negative singular parts give
  $S_1$ and $S_2$. The uniqueness statement follows by comparing zero sets,
  boundary moduli, and singular measures.
\end{proof}
